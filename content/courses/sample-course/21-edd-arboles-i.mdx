# Estructuras de Datos Avanzadas · Árboles I

Un arreglo, una lista, una pila y una cola guardan sus elementos en fila:
cada elemento tiene a lo más un siguiente. Muchos datos no tienen esa forma.

{/* HierarchyPicture — a static drawing of a general tree (any number of children per node): the folder tree or a small hierarchy, with tones, badges, notes, level lines, an open path and an optional queue under it. */}
{/* File-local on purpose (add-a-course-document.md §6e-quater): one consumer, no catalog entry, not registered. The second document that wants it promotes it. */}
{/* A leaf takes the next column; a parent sits centred over its first and last child. Every node name in one tree is unique, because tones, badges and notes are keyed by it. */}
{/* edgeLabels writes a word beside the edge that reaches a node, keyed by that node's name: the answer a decision takes to get there. */}
{/* minWidth gives figures set side by side one canvas width, so their boxes and text come out the same size; the tree is centred in it. compact and size fit a frame of a step-through, whose box has a fixed height. A node toned "hidden" is not drawn, nor its edge, but keeps its place: a tree built node by node grows in the positions it will end in, and nothing moves between frames. fresh names the node just added; its edge is drawn in the mark colour. */}
{/* The drawings of an order, each in its own colour, and each told by its arrows and numbers, never by colour alone. flow puts an arrowhead on every edge — "down" (focus) for what a node takes from its parent, "up" (mark) for what it takes from its children. visits numbers the nodes in "pre" (focus) or "post" (mark) order and draws an arrow from every visit to the next: along the edge when the next is a child or the parent, dashed when the walk jumps to another branch. reading numbers them level by level (keep) and draws each level as a line read left to right, with the return to the next. strip writes the sequence under the tree, the order a program would print it in. */}

export function HierarchyPicture({ tree, tones = {}, badges = {}, notes = {}, numbers = {}, path = [], levels = false, queue, caption, minWidth = 0, flow, visits, reading = false, strip = false, badgeColor = 'var(--color-focus)', numbersColor = 'var(--color-keep)', compact = false, size = 1, fresh, ruler, edgeLabels = {}, label }) {
  const boxH = 26;
  const levelH = compact ? 50 : 64;
  const padX = reading ? 22 : 12;
  const padTop = visits || reading ? 26 : 20;
  // ruler draws the tree's height at the left: a line from the root's level to the last one, a tick per level, and its label ("h = 3").
  const labelW = (levels || reading ? 62 : 0) + (ruler ? 48 : 0);
  const widthOf = (name) => Math.max(40, name.length * 7.6 + 16);
  const nodes = [];
  let leaves = 0;
  const walk = (t, depth, parent) => {
    const me = { name: t.n, depth, parent, kids: [] };
    nodes.push(me);
    if (t.c && t.c.length > 0) {
      for (const c of t.c) me.kids.push(walk(c, depth + 1, me));
      me.x = (me.kids[0].x + me.kids[me.kids.length - 1].x) / 2;
    } else {
      me.x = leaves;
      leaves += 1;
    }
    return me;
  };
  walk(tree, 0, null);
  const pitch = Math.max(...nodes.map((n) => widthOf(n.name))) + 14;
  const depth = Math.max(...nodes.map((n) => n.depth));
  const hasNotes = Object.keys(notes).length > 0;
  const treeH = padTop + boxH + depth * levelH + (hasNotes ? 18 : 6);
  // The visit sequence, from the recursion itself for the two orders and level by level for the reading.
  const pre = [];
  const post = [];
  const recurse = (n) => {
    pre.push(n);
    n.kids.forEach(recurse);
    post.push(n);
  };
  recurse(nodes[0]);
  const byLevel = [...Array(depth + 1).keys()].map((d) => nodes.filter((n) => n.depth === d).sort((a, b) => a.x - b.x));
  const seq = visits === 'pre' ? pre : visits === 'post' ? post : reading ? byLevel.flat() : [];
  const order = {};
  seq.forEach((n, i) => {
    order[n.name] = i + 1;
  });
  const seqColor = visits === 'post' ? 'var(--color-mark)' : visits === 'pre' ? 'var(--color-focus)' : 'var(--color-keep)';
  const flowColor = flow === 'up' ? 'var(--color-mark)' : 'var(--color-focus)';
  // The strip is one row, or two even ones past six chips: chip, arrow, chip. A row wider than the drawing widens the canvas.
  const drawingW = labelW + 2 * padX + leaves * pitch;
  const chipW = (n) => widthOf(n.name) + 20;
  const gap = 22;
  const perRow = seq.length > 6 ? Math.ceil(seq.length / 2) : seq.length;
  const rows = strip ? [seq.slice(0, perRow), seq.slice(perRow)].filter((r) => r.length > 0) : [];
  const rowWidth = (row) => row.reduce((w, n, i) => w + chipW(n) + (i > 0 ? gap : 0), 0);
  const stripW = Math.max(0, ...rows.map(rowWidth));
  const stripY = treeH + 16;
  const stripH = strip ? rows.length * (boxH + 12) : 0;
  const queueY = treeH + 26;
  const captionY = queue ? queueY + boxH + 26 : strip ? stripY + stripH + 10 : treeH + 18;
  // A caption wider than the drawing widens the canvas, and the drawing is centred in it, so no caption is ever clipped.
  const width = Math.max(drawingW, minWidth, caption ? caption.length * 6.6 + 24 : 0, stripW + 2 * padX);
  const offset = (width - drawingW) / 2;
  const height = captionY + (caption ? 8 : -6);
  const cx = (n) => offset + labelW + padX + (n.x + 0.5) * pitch;
  const cy = (n) => padTop + boxH / 2 + n.depth * levelH;
  const onPath = new Set(path);
  const TONE = {
    idle: ['var(--color-surface)', 'var(--color-rule-strong)', 1.3],
    focus: ['var(--color-surface)', 'var(--color-focus)', 2.4],
    mark: ['var(--color-mark-soft)', 'var(--color-mark)', 2.2],
    keep: ['var(--color-keep-soft)', 'var(--color-keep)', 2.2],
    done: ['var(--color-sunk)', 'var(--color-rule-strong)', 1.3],
    faint: ['var(--color-surface)', 'var(--color-rule)', 1.2],
  };
  const deg = (dy, dx) => (Math.atan2(dy, dx) * 180) / Math.PI;
  const head = (x, y, angle, color, key) => (
    <polygon key={key} points="0,0 -8,-4.5 -8,4.5" transform={`translate(${x} ${y}) rotate(${angle})`} style={{ fill: color }} />
  );
  const dot = (x, y, k, color, key) => (
    <g key={key}>
      <circle cx={x} cy={y} r="9" style={{ fill: color }} />
      <text x={x} y={y + 3.8} textAnchor="middle" fontSize="11" fontWeight="700" style={{ fill: 'var(--color-ground)' }}>{k}</text>
    </g>
  );
  // An edge with its arrowhead at the middle, pointing to the child (down) or to the parent (up).
  const edgeArrow = (parent, child, up, color, key, width = 1.8) => {
    const x1 = cx(parent);
    const y1 = cy(parent) + boxH / 2;
    const x2 = cx(child);
    const y2 = cy(child) - boxH / 2;
    const a = deg(y2 - y1, x2 - x1);
    const r = (a * Math.PI) / 180;
    const s = up ? -4 : 4;
    return (
      <g key={key}>
        <line x1={x1} y1={y1} x2={x2} y2={y2} style={{ stroke: color }} strokeWidth={width} />
        {head((x1 + x2) / 2 + s * Math.cos(r), (y1 + y2) / 2 + s * Math.sin(r), up ? a + 180 : a, color, `${key}-h`)}
      </g>
    );
  };
  // From one visit to the next: along the edge when they are parent and child; otherwise a dashed jump that leaves the right of one and enters the left of the other, both ends level, so the arrowhead points the way the line arrives.
  const step = (a, b, key) => {
    if (b.parent === a) return edgeArrow(a, b, false, seqColor, key, 2.4);
    if (a.parent === b) return edgeArrow(b, a, true, seqColor, key, 2.4);
    const sx = cx(a) + widthOf(a.name) / 2 + 2;
    const sy = cy(a) + 3;
    const ex = cx(b) - widthOf(b.name) / 2 - 2;
    const ey = cy(b) + 3;
    const d = Math.max(22, (ex - sx) / 2);
    return (
      <g key={key}>
        <path d={`M ${sx} ${sy} C ${sx + d} ${sy}, ${ex - d} ${ey}, ${ex - 7} ${ey}`} fill="none" style={{ stroke: seqColor }} strokeWidth="1.8" strokeDasharray="5 4" />
        {head(ex, ey, 0, seqColor, `${key}-h`)}
      </g>
    );
  };
  const shown = nodes.filter((n) => tones[n.name] !== 'hidden');
  const edges = shown.filter((n) => n.parent !== null);
  return (
    <svg viewBox={`0 0 ${width} ${height}`} style={{ width: '100%', maxWidth: `${Math.round(width * 1.35 * size)}px`, height: 'auto', display: 'block', margin: compact ? '0.25rem auto' : '1rem auto' }} role="img" aria-label={label}>
      {ruler ? (
        <g>
          <line x1={offset + 26} y1={padTop + boxH / 2} x2={offset + 26} y2={padTop + boxH / 2 + depth * levelH} style={{ stroke: 'var(--color-ink)' }} strokeWidth="1.8" />
          {[...Array(depth + 1).keys()].map((d) => (
            <line key={`rt${d}`} x1={offset + 20} y1={padTop + boxH / 2 + d * levelH} x2={offset + 32} y2={padTop + boxH / 2 + d * levelH} style={{ stroke: 'var(--color-ink)' }} strokeWidth="1.8" />
          ))}
          <text x={offset + 14} y={padTop + boxH / 2 + (depth * levelH) / 2} textAnchor="middle" fontSize="13" fontWeight="700" transform={`rotate(-90 ${offset + 14} ${padTop + boxH / 2 + (depth * levelH) / 2})`} style={{ fill: 'var(--color-ink)' }}>{ruler}</text>
        </g>
      ) : null}
      {levels && !reading
        ? [...new Set(nodes.map((n) => n.depth))].map((d) => (
            <g key={`lv${d}`}>
              <line x1={offset + labelW - 4} y1={padTop + boxH / 2 + d * levelH} x2={offset + drawingW - 4} y2={padTop + boxH / 2 + d * levelH} style={{ stroke: 'var(--color-rule)' }} strokeWidth="1" strokeDasharray="3 5" />
              <text x={offset + (ruler ? 52 : 4)} y={padTop + boxH / 2 + d * levelH + 4} fontSize="11" style={{ fill: 'var(--color-ink-faint)' }}>{`nivel ${d}`}</text>
            </g>
          ))
        : null}
      {reading
        ? byLevel.map((row, d) => {
            const y = padTop + boxH / 2 + d * levelH;
            const x1 = cx(row[0]) - widthOf(row[0].name) / 2 - 14;
            const x2 = cx(row[row.length - 1]) + widthOf(row[row.length - 1].name) / 2 + 16;
            const next = byLevel[d + 1];
            const nx = next ? cx(next[0]) - widthOf(next[0].name) / 2 - 14 : 0;
            return (
              <g key={`rd${d}`}>
                <text x={offset + 4} y={y + 4} fontSize="11" style={{ fill: 'var(--color-ink-faint)' }}>{`nivel ${d}`}</text>
                <line x1={x1} y1={y} x2={x2 - 2} y2={y} style={{ stroke: 'var(--color-keep)' }} strokeWidth="2" />
                {head(x2, y, 0, 'var(--color-keep)', `rh${d}`)}
                {next ? (
                  <g>
                    <path d={`M ${x2} ${y + 6} C ${x2 + 10} ${y + levelH * 0.55}, ${nx - 30} ${y + levelH * 0.35}, ${nx - 6} ${y + levelH - 6}`} fill="none" style={{ stroke: 'var(--color-keep)' }} strokeWidth="1.5" strokeDasharray="4 4" />
                    {head(nx - 2, y + levelH - 2, 45, 'var(--color-keep)', `rr${d}`)}
                  </g>
                ) : null}
              </g>
            );
          })
        : null}
      {edges.map((n) => {
        if (flow) return edgeArrow(n.parent, n, flow === 'up', flowColor, `e-${n.name}`);
        const onIt = (onPath.has(n.name) && onPath.has(n.parent.name)) || n.name === fresh;
        const color = n.name === fresh ? 'var(--color-mark)' : onIt ? 'var(--color-focus)' : tones[n.name] === 'faint' ? 'var(--color-rule)' : 'var(--color-rule-strong)';
        return <line key={`e-${n.name}`} x1={cx(n.parent)} y1={cy(n.parent) + boxH / 2} x2={cx(n)} y2={cy(n) - boxH / 2} style={{ stroke: color }} strokeWidth={onIt ? 2.6 : 1.2} />;
      })}
      {edges
        .filter((n) => edgeLabels[n.name] !== undefined)
        .map((n) => {
          const left = cx(n) < cx(n.parent);
          return (
            <text key={`el-${n.name}`} x={(cx(n.parent) + cx(n)) / 2 + (left ? -8 : 8)} y={(cy(n.parent) + cy(n)) / 2 + 4} textAnchor={left ? 'end' : 'start'} fontSize="12" fontWeight="700" paintOrder="stroke" strokeWidth="3" style={{ fill: 'var(--color-ink-soft)', stroke: 'var(--color-ground)' }}>{edgeLabels[n.name]}</text>
          );
        })}
      {visits ? seq.slice(1).map((b, i) => step(seq[i], b, `s-${b.name}`)) : null}
      {shown.length === 0 ? (
        <text x={cx(nodes[0])} y={cy(nodes[0]) + 4.5} textAnchor="middle" fontSize="13" fontFamily="var(--font-mono, monospace)" style={{ fill: 'var(--color-ink-soft)' }}>root = null</text>
      ) : null}
      {shown.map((n) => {
        const [fill, stroke, weight] = TONE[tones[n.name] ?? 'idle'];
        const w = widthOf(n.name);
        const faint = tones[n.name] === 'faint';
        return (
          <g key={`n-${n.name}`} opacity={faint ? 0.5 : undefined}>
            <rect x={cx(n) - w / 2} y={cy(n) - boxH / 2} width={w} height={boxH} rx="6" style={{ fill, stroke }} strokeWidth={weight} strokeDasharray={faint ? '4 3' : undefined} />
            <text x={cx(n)} y={cy(n) + 4.5} textAnchor="middle" fontSize="13" fontWeight="600" style={{ fill: 'var(--color-ink)' }}>{n.name}</text>
            {badges[n.name] !== undefined ? (
              <text x={cx(n) + w / 2 - 3} y={cy(n) - boxH / 2 - 3} fontSize="11" fontWeight="700" paintOrder="stroke" strokeWidth="3" style={{ fill: badgeColor, stroke: 'var(--color-ground)' }}>{badges[n.name]}</text>
            ) : null}
            {/* A leaf's note sits under it; a parent's goes to its left, clear of the edges leaving for its children. */}
            {notes[n.name] !== undefined ? (
              n.kids.length === 0 ? (
                <text x={cx(n)} y={cy(n) + boxH / 2 + 13} textAnchor="middle" fontSize="10.5" style={{ fill: 'var(--color-ink-soft)' }}>{notes[n.name]}</text>
              ) : (
                <text x={cx(n) - w / 2 - 5} y={cy(n) + 4} textAnchor="end" fontSize="10.5" paintOrder="stroke" strokeWidth="3" style={{ fill: 'var(--color-ink-soft)', stroke: 'var(--color-ground)' }}>{notes[n.name]}</text>
              )
            ) : null}
          </g>
        );
      })}
      {seq.map((n) => dot(cx(n) - widthOf(n.name) / 2, cy(n) - boxH / 2, order[n.name], seqColor, `sd-${n.name}`))}
      {nodes.filter((n) => numbers[n.name] !== undefined).map((n) => dot(cx(n) - widthOf(n.name) / 2, cy(n) - boxH / 2, numbers[n.name], numbersColor, `nd-${n.name}`))}
      {strip
        ? rows.map((row, r) => {
            let x = (width - rowWidth(row)) / 2;
            const y = stripY + r * (boxH + 12);
            return row.map((n, i) => {
              const w = chipW(n);
              const x0 = x + (i > 0 ? gap : 0);
              x = x0 + w;
              return (
                <g key={`c-${n.name}`}>
                  {i > 0 ? <line x1={x0 - gap + 4} y1={y + boxH / 2} x2={x0 - 6} y2={y + boxH / 2} style={{ stroke: seqColor }} strokeWidth="1.6" /> : null}
                  {i > 0 ? head(x0 - 3, y + boxH / 2, 0, seqColor, `ch-${n.name}`) : null}
                  <rect x={x0} y={y} width={w} height={boxH} rx="13" style={{ fill: 'var(--color-surface)', stroke: seqColor }} strokeWidth="1.6" />
                  {dot(x0 + 13, y + boxH / 2, order[n.name], seqColor, `cd-${n.name}`)}
                  <text x={x0 + 26 + (w - 26) / 2} y={y + boxH / 2 + 4.5} textAnchor="middle" fontSize="13" fontWeight="600" style={{ fill: 'var(--color-ink)' }}>{n.name}</text>
                </g>
              );
            });
          })
        : null}
      {queue ? (
        <g>
          <text x={offset + labelW + padX} y={queueY - 8} fontSize="11" style={{ fill: 'var(--color-ink-faint)' }}>{queue.length === 0 ? 'cola · vacía' : 'cola · sale por la izquierda'}</text>
          {queue.map((name, i) => {
            const w = pitch - 14;
            const x = offset + labelW + padX + i * pitch;
            return (
              <g key={`q-${name}`}>
                <rect x={x} y={queueY} width={w} height={boxH} rx="3" style={{ fill: 'var(--color-mark-soft)', stroke: 'var(--color-mark)' }} strokeWidth="1.6" />
                <text x={x + w / 2} y={queueY + 17.5} textAnchor="middle" fontSize="13" fontWeight="600" style={{ fill: 'var(--color-ink)' }}>{name}</text>
              </g>
            );
          })}
        </g>
      ) : null}
      {caption ? (
        <text x={width / 2} y={captionY} textAnchor="middle" fontSize="12" style={{ fill: 'var(--color-ink-soft)' }}>{caption}</text>
      ) : null}
    </svg>
  );
}

{/* OrderRule — the rule of a traversal written short, as pills joined by arrows in the traversal's colour: nodo → hijos for the preorder, the colour its drawings use. File-local like HierarchyPicture (§6e-quater). */}

export function OrderRule({ parts, tone = 'focus', label }) {
  const color = `var(--color-${tone})`;
  return (
    <div role="img" aria-label={label} style={{ display: 'flex', flexWrap: 'wrap', justifyContent: 'center', alignItems: 'center', gap: '0.6rem', margin: '1rem 0', fontWeight: 700, fontSize: '1.15em' }}>
      {parts.flatMap((part, i) => [
        i > 0 ? <span key={`a${i}`} aria-hidden="true" style={{ color }}>→</span> : null,
        <span key={`p${i}`} aria-hidden="true" style={{ border: `2px solid ${color}`, borderRadius: '999px', padding: '0.1rem 0.9rem', color: 'var(--color-ink)' }}>{part}</span>,
      ])}
    </div>
  );
}

{/* TreeMemory — a general tree as it sits in memory, in the linked-list class's idiom: a node is a box with its value and a grey compartment whose dot is the reference to the sequence of its children, an ArrayList, drawn as the row of cells of the dynamic array behind it, one per child, each with the dot of its reference and an arrow to the child. A leaf's list is empty and drawn as Java prints an empty ArrayList, []. depth cuts the drawing: a node at the cut keeps its list, with short faint arrows to children not drawn. entry draws the Tree object and its root reference ("root", or "null" for the empty tree); fields names the node's two fields. A node toned "mark" is the one just added, and so is its cell. File-local like HierarchyPicture (§6e-quater). */}

export function TreeMemory({ tree, depth = 99, tones = {}, entry, fields = false, size = 1, minWidth = 0, px, caption, label }) {
  const H = 28;
  const COMP = 20;
  const CELL = 22;
  const GAP = 18;
  const LEVEL = 104;
  const pad = 14;
  const top = pad + (fields ? 16 : 0);
  const vw = (name) => Math.max(44, name.length * 7.6 + 14);
  const nodeW = (t) => vw(t.n) + COMP;
  const kidsOf = (t) => t.c ?? [];
  const drawn = (t, d) => (d + 1 < depth ? kidsOf(t) : []);
  const listW = (t) => (kidsOf(t).length === 0 ? 30 : kidsOf(t).length * CELL);
  const span = (t, d) => {
    const inner = drawn(t, d).reduce((s, k, i) => s + span(k, d + 1) + (i > 0 ? GAP : 0), 0);
    return Math.max(nodeW(t) + 8, listW(t) + 8, inner);
  };
  const entryW = entry ? 104 : 0;
  const placed = [];
  const place = (t, d, x0, from) => {
    const sp = span(t, d);
    const cx = x0 + sp / 2;
    const y = top + d * LEVEL;
    const me = { t, d, cx, y, from, cut: d + 1 >= depth && kidsOf(t).length > 0 };
    placed.push(me);
    const lw = listW(t);
    me.lx = cx - lw / 2;
    me.ly = y + H + 18;
    const ks = drawn(t, d);
    const inner = ks.reduce((s, k, i) => s + span(k, d + 1) + (i > 0 ? GAP : 0), 0);
    let x = cx - inner / 2;
    ks.forEach((k, i) => {
      place(k, d + 1, x, { x: me.lx + i * CELL + CELL / 2, y: me.ly + CELL / 2 });
      x += span(k, d + 1) + GAP;
    });
  };
  place(tree, 0, pad + entryW, null);
  const deepest = Math.max(...placed.map((p) => p.d));
  const stubs = placed.some((p) => p.cut);
  // minWidth gives drawings set side by side one canvas width, so their boxes come out the same size.
  const width = Math.max(2 * pad + entryW + span(tree, 0), minWidth, caption ? caption.length * 6.6 + 24 : 0);
  const shift = (width - (2 * pad + entryW + span(tree, 0))) / 2;
  const bodyH = top + deepest * LEVEL + H + 18 + CELL + (stubs ? 26 : 0) + pad;
  const height = bodyH + (caption ? 18 : 0);
  const arrow = (x1, y1, x2, y2, key, color = 'var(--color-ink-soft)', faint = false) => {
    const a = Math.atan2(y2 - y1, x2 - x1);
    const h = 7;
    const tip = (s) => `${x2 - h * Math.cos(a + s)},${y2 - h * Math.sin(a + s)}`;
    const stroke = faint ? 'var(--color-rule)' : color;
    return (
      <g key={key}>
        <line x1={x1} y1={y1} x2={x2 - 5 * Math.cos(a)} y2={y2 - 5 * Math.sin(a)} style={{ stroke }} strokeWidth="1.6" />
        <path d={`M ${x2},${y2} L ${tip(0.42)} L ${tip(-0.42)} z`} style={{ fill: stroke }} />
      </g>
    );
  };
  const root = placed[0];
  return (
    // px draws at a fixed scale, its own width times px, so drawings of different sizes set in one row share a scale.
    <svg viewBox={`0 0 ${width} ${height}`} style={{ width: px ? `${Math.round(width * px)}px` : '100%', maxWidth: px ? '100%' : `${Math.round(width * 1.3 * size)}px`, height: 'auto', display: 'block', margin: '0.75rem auto' }} role="img" aria-label={label}>
      <g transform={`translate(${shift} 0)`}>
        {entry ? (
          <g>
            <text x={pad} y={root.y - 6} fontSize="11" style={{ fill: 'var(--color-ink-faint)' }}>Tree</text>
            <rect x={pad} y={root.y} width="52" height={H} rx="4" style={{ fill: 'var(--color-surface)', stroke: 'var(--color-rule-strong)' }} strokeWidth="1.6" />
            <text x={pad + 26} y={root.y + 18} textAnchor="middle" fontSize="12" fontFamily="var(--font-mono, monospace)" style={{ fill: 'var(--color-ink)' }}>root</text>
            <rect x={pad + 52} y={root.y} width={COMP} height={H} rx="3" style={{ fill: 'var(--color-sunk)', stroke: 'var(--color-rule-strong)' }} strokeWidth="1.6" />
            <circle cx={pad + 52 + COMP / 2} cy={root.y + H / 2} r="2.6" style={{ fill: 'var(--color-ink-soft)' }} />
            {entry === 'null' ? (
              <g>

            )}
          </g>
        ) : null}
        {entry === 'null'
          ? null

              return (
                <g key={`m-${t.n}`}>
                  {fields && p.d === 0 ? (
                    <g>
                      <text x={x + w / 2} y={p.y - 5} textAnchor="middle" fontSize="10" fontFamily="var(--font-mono, monospace)" style={{ fill: 'var(--color-ink-faint)' }}>value</text>

                    </g>
                  ) : null}
                  {p.from ? arrow(p.from.x, p.from.y, p.cx, p.y - 1, `a-${t.n}`, isNew ? 'var(--color-mark)' : undefined) : null}
                  <rect x={x} y={p.y} width={w} height={H} rx="4" style={{ fill: isNew ? 'var(--color-mark-soft)' : 'var(--color-surface)', stroke }} strokeWidth={isNew ? 2.4 : 1.6} />
                  <rect x={x + w} y={p.y} width={COMP} height={H} rx="3" style={{ fill: 'var(--color-sunk)', stroke }} strokeWidth={isNew ? 2.4 : 1.6} />

                      <rect x={p.lx} y={p.ly} width="30" height={CELL} rx="3" style={{ fill: 'var(--color-sunk)', stroke: isNew ? 'var(--color-mark)' : 'var(--color-rule)' }} strokeWidth="1.2" strokeDasharray="3 3" />

                    </g>
                  ) : (
                    kids.map((k, i) => {
                      const cellNew = tones[k.n] === 'mark';
                      const cx = p.lx + i * CELL + CELL / 2;
                      return (
                        <g key={`c-${k.n}`}>
                          <rect x={p.lx + i * CELL} y={p.ly} width={CELL} height={CELL} rx="2" style={{ fill: cellNew ? 'var(--color-mark-soft)' : 'var(--color-sunk)', stroke: cellNew ? 'var(--color-mark)' : 'var(--color-rule-strong)' }} strokeWidth={cellNew ? 2.2 : 1.3} />
                          <circle cx={cx} cy={p.ly + CELL / 2} r="2.6" style={{ fill: 'var(--color-ink-soft)' }} />
                          {p.cut ? arrow(cx, p.ly + CELL / 2, cx, p.ly + CELL + 20, `s-${k.n}`, undefined, true) : null}

                      );
                    })
                  )}
                </g>
              );
            })}
      </g>
      {caption ? (
        <text x={width / 2} y={bodyH + 6} textAnchor="middle" fontSize="12" style={{ fill: 'var(--color-ink-soft)' }}>{caption}</text>
      ) : null}
    </svg>
  );
}

<Slide title="Jerarquías">
Muchos datos se organizan por **niveles**: cada elemento tiene **uno solo
encima** y puede tener **varios debajo**. Las carpetas de un computador son
así, y también un organigrama, una expresión aritmética y un árbol
genealógico.

Esa forma se llama **jerarquía**, y la estructura de datos que la representa
es el **árbol**. El nombre viene del dibujo, que es un árbol al revés: la
**raíz** queda arriba y las **hojas**, abajo.

</Slide>

<Slide title="El árbol y sus nodos">
Un **árbol** es una colección de elementos, llamados **nodos**, organizados
en una jerarquía:

1. **Un nodo, la raíz, está arriba de todo** y no cuelga de ningún otro.
2. **Cada uno de los demás cuelga de exactamente un nodo**, su **padre**.
   Los nodos que cuelgan de un mismo padre son sus **hijos**.

La línea que une a un padre con un hijo se llama **arista**. Un nodo puede
tener cualquier cantidad de hijos, incluso ninguno. Cada nodo, salvo la raíz,
tiene exactamente una arista, la que lo une con su padre: por eso **un árbol
de $$n$$ nodos tiene $$n - 1$$ aristas**.

<HierarchyPicture tree={FOLDERS} tones={{ 'home/': 'focus', 'ramos/': 'mark', 'eda/': 'keep', 'calculo/': 'keep' }} badges={{ 'home/': 'raíz', 'ramos/': 'padre' }} notes={{ 'eda/': 'hijo', 'calculo/': 'hijo' }} label="El árbol de carpetas con la raíz home marcada, y la carpeta ramos marcada como padre de sus dos hijos, eda y calculo." caption="home/ es la raíz; ramos/ es el padre de eda/ y calculo/ · 11 nodos, 10 aristas" />

</Slide>

<Slide title="Hojas y nodos internos">
Un nodo sin hijos es una **hoja**. Un nodo con al menos un hijo es un
**nodo interno**. En un sistema de archivos, las hojas son los archivos y las
carpetas vacías, y los nodos internos son las carpetas que guardan algo.

<HierarchyPicture tree={FOLDERS} tones={{ 'playa.jpg': 'mark', 'perro.jpg': 'mark', 't1.pdf': 'mark', 't2.pdf': 'mark', 'guia.pdf': 'mark', 'notas.txt': 'mark' }} notes={{ 'playa.jpg': 'hoja', 'perro.jpg': 'hoja', 't1.pdf': 'hoja', 't2.pdf': 'hoja', 'guia.pdf': 'hoja', 'notas.txt': 'hoja' }} label="El árbol de carpetas con sus seis hojas marcadas: playa.jpg, perro.jpg, t1.pdf, t2.pdf, guia.pdf y notas.txt. Las cinco carpetas son nodos internos." caption="seis hojas, los archivos; cinco nodos internos, las carpetas" />

</Slide>

<Slide title="Caminos desde la raíz">
Un **camino** es una secuencia de nodos en la que cada uno es el padre del
siguiente. Su **largo** es la cantidad de aristas que recorre: uno menos que
la cantidad de nodos.

Desde la raíz hasta cualquier nodo hay **exactamente un camino**, porque
cada nodo tiene un solo padre y subir de padre en padre no deja opciones. En
un sistema de archivos, ese camino es la **ruta** del archivo.

<HierarchyPicture tree={FOLDERS} tones={{ 'home/': 'focus', 'ramos/': 'focus', 'eda/': 'focus', 't1.pdf': 'focus' }} path={['home/', 'ramos/', 'eda/', 't1.pdf']} label="El camino desde la raíz hasta t1.pdf: home, ramos, eda, t1.pdf. Cuatro nodos y tres aristas." caption="home/ramos/eda/t1.pdf: cuatro nodos, un camino de largo 3" />

</Slide>

<Slide title="Ancestros, descendientes y hermanos">
Los nodos del camino desde la raíz hasta un nodo, sin contarlo a él, son sus
**ancestros**, y él es **descendiente** de cada uno de ellos. Los nodos que
tienen el mismo padre son **hermanos**.

<HierarchyPicture tree={FOLDERS} tones={{ 'home/': 'focus', 'ramos/': 'focus', 'eda/': 'focus', 't1.pdf': 'mark', 't2.pdf': 'keep' }} path={['home/', 'ramos/', 'eda/', 't1.pdf']} notes={{ 'home/': 'ancestro', 'ramos/': 'ancestro', 'eda/': 'ancestro', 't2.pdf': 'hermano' }} label="Los ancestros de t1.pdf son home, ramos y eda, los nodos de su camino desde la raíz. Su hermano es t2.pdf, que tiene el mismo padre, eda." caption="ancestros de t1.pdf: home/, ramos/ y eda/ · su hermano: t2.pdf" />

</Slide>

<Slide title="Subárboles">
Un nodo, junto con todos sus descendientes, forma un árbol que tiene a ese
nodo como raíz: un **subárbol**. Cada carpeta
es la raíz de un subárbol, el de todo lo que guarda.

<HierarchyPicture tree={FOLDERS} tones={{ 'home/': 'faint', 'fotos/': 'faint', 'playa.jpg': 'faint', 'perro.jpg': 'faint', 'notas.txt': 'faint', 'ramos/': 'keep', 'eda/': 'keep', 't1.pdf': 'keep', 't2.pdf': 'keep', 'calculo/': 'keep', 'guia.pdf': 'keep' }} label="El subárbol que tiene raíz en ramos: ramos, eda, t1.pdf, t2.pdf, calculo y guia.pdf. El resto del árbol aparece atenuado." caption="el subárbol de ramos/: seis nodos, y ramos/ es su raíz" />

Por eso un árbol se puede definir a partir de árboles más pequeños: **un árbol
es una raíz y, debajo de ella, cero o más subárboles**. Esa definición es
**recursiva**, y las operaciones sobre árboles también lo son: lo que se
hace con el árbol se hace, igual, con cada subárbol.

</Slide>

<Slide title="Profundidad y niveles">
La **profundidad** de un nodo es el largo del camino desde la raíz hasta él.
La raíz tiene profundidad 0, sus hijos 1, los hijos de ellos 2, y así.

Los nodos de igual profundidad forman un **nivel**. El nivel $$d$$ reúne
a todos los nodos que están a $$d$$ aristas de la raíz.

<HierarchyPicture tree={FOLDERS} levels label="El árbol de carpetas con una línea por nivel. Nivel 0: home. Nivel 1: fotos, ramos y notas.txt. Nivel 2: playa.jpg, perro.jpg, eda y calculo. Nivel 3: t1.pdf, t2.pdf y guia.pdf." caption="la profundidad de cada nodo es el número de su nivel" />

</Slide>

<Slide title="La altura">
La **altura** de un nodo es el largo del camino más largo que baja desde él
hasta una hoja. Una hoja tiene altura 0, y un nodo interno tiene una unidad
más que el más alto de sus hijos.

La **altura del árbol**, que se escribe $$h$$, es la altura de su raíz: la
cantidad de niveles menos uno.

<HierarchyPicture tree={FOLDERS} ruler="h = 3" badges={{ 'home/': 'h=3', 'fotos/': 'h=1', 'playa.jpg': 'h=0', 'perro.jpg': 'h=0', 'ramos/': 'h=2', 'eda/': 'h=1', 't1.pdf': 'h=0', 't2.pdf': 'h=0', 'calculo/': 'h=1', 'guia.pdf': 'h=0', 'notas.txt': 'h=0' }} label="El árbol de carpetas con la altura de cada nodo y, a la izquierda, una línea de la raíz al último nivel rotulada h = 3: tres aristas, una por nivel. La altura dice cuántas aristas hay que bajar, como máximo, desde la raíz para
llegar a cualquier nodo.

</Slide>

<Slide title="Desde arriba y desde abajo">
La profundidad de un nodo sale de su **padre**: es la del padre más uno. Se
conoce bajando desde la raíz, y cada nodo la sabe **antes** que sus hijos.

La altura de un nodo sale de sus **hijos**: es la de su hijo más alto, más
uno. Se conoce subiendo desde las hojas, y cada nodo la sabe **después** que
sus hijos.

## La clase árbol

<Slide title="El nodo y el árbol">
Cada nodo guarda su valor y la **secuencia de sus hijos**.

A una lista enlazada se entra por `head`; a un árbol, por `root`, la **raíz**.
Desde ella se llega a cualquier nodo.

<Slide title="Agregar un hijo">
`add` es un método de `Node`: crea el hijo, lo pone al **final** de la
secuencia y lo devuelve, para poder seguir agregando debajo de él. El orden de
esa secuencia es el orden de los hijos de izquierda a derecha, el que siguen
los recorridos. Agregar al final de un `ArrayList` cuesta $$\Theta(1)$$
amortizado.

```java
Node<T> add(T value) {
    Node<T> child = new Node<>(value);
    children.add(child);
    return child;
}
```

## Recorridos de un árbol

<Slide title="El recorrido de un árbol">
**Recorrer** una estructura es visitar cada uno de sus elementos una vez. En
una fila hay un orden natural: del primero al último. En un árbol no lo hay:
después de la raíz puede seguir cualquiera de sus hijos, y después de un
hijo, uno de sus hijos o uno de sus hermanos.

Un **recorrido** es una regla que decide ese orden. Hay varias, y cada una
responde una pregunta distinta sobre la jerarquía.

</Slide>

<Slide title="El preorden">
En **preorden**, cada nodo se visita **antes** que sus hijos, y los hijos se
recorren de izquierda a derecha, cada uno con la misma regla:

<OrderRule parts={['nodo', 'hijos']} label="La regla del preorden: el nodo, y después sus hijos." />

Es el orden de la profundidad: un nodo se visita antes que todo lo que cuelga
de él.

<HierarchyPicture tree={FOLDERS} visits="pre" strip label="El árbol de carpetas con el preorden: cada nodo numerado en el orden en que se visita, y flechas de cada visita a la siguiente, que bajan por las aristas y saltan al hermano siguiente cuando un subárbol se termina. home 1, fotos 2, playa.jpg 3, perro.jpg 4, ramos 5, eda 6, t1.pdf 7, t2.pdf 8, calculo 9, guia.pdf 10 y notas.txt 11. Debajo, la misma secuencia en fila." caption="las flechas siguen la visita; abajo, el orden en que se visitan" />

</Slide>

<Slide title="El código del preorden">
La regla **nodo → hijos** se escribe en dos partes: primero la **visita**, lo
que se hace con cada nodo, y después el **recorrido**, un `for` que llama al
mismo método con cada hijo.

El recorrido es siempre el mismo; lo que cambia según el problema es la
visita. Un explorador de archivos, o el comando `tree` de una terminal, lista
una carpeta en preorden: la visita imprime una línea con la sangría de la
**profundidad** del nodo, y cada hijo recibe la de su padre, más uno.

<Slide title="El postorden">
En **postorden**, cada nodo se visita **después** de sus hijos: primero se
recorren los hijos, de izquierda a derecha y cada uno con la misma regla, y
al final el nodo.

<OrderRule parts={['hijos', 'nodo']} tone="mark" label="La regla del postorden: los hijos, y después el nodo." />

Es el orden de la altura: un nodo se visita después de todo lo que cuelga de
él.

</Slide>

<Slide title="El recorrido por niveles">
El recorrido **por niveles** visita primero la raíz, después todos los nodos
de profundidad 1, después los de profundidad 2, y así; dentro de cada nivel,
de izquierda a derecha. Se lee como un texto, en el que cada nivel es un
renglón:

<OrderRule parts={['nivel 0', 'nivel 1', 'nivel 2', '…']} tone="keep" label="La regla del recorrido por niveles: el nivel 0, después el 1, después el 2, y así, cada uno de izquierda a derecha." />

Es el orden de la distancia a la raíz: un nodo se visita antes que todos los
que están más lejos de ella.

<HierarchyPicture tree={FOLDERS} reading strip label="El árbol de carpetas leído por niveles, como un texto: cada nivel de izquierda a derecha, y después el comienzo del siguiente. home 1; fotos 2, ramos 3, notas.txt 4; playa.jpg 5, perro.jpg 6, eda 7, calculo 8; t1.pdf 9, t2.pdf 10 y guia.pdf 11." caption="cada renglón de izquierda a derecha; abajo, el orden en que se visitan" />

</Slide>

<Slide title="Una cola para recorrer por niveles">
El preorden y el postorden siguen la recursión que define al árbol: un nodo
y, debajo, sus subárboles. El recorrido por niveles no sigue esa forma,
porque salta de un subárbol a otro dentro del mismo nivel. Lo que lo ordena
es una **cola**, que entrega los elementos en el mismo orden en que entraron.

Un nodo entra a la cola cuando se visita su padre, así que los hijos de los
nodos de un nivel entran todos detrás de ese nivel: la cola tiene siempre lo
que falta del nivel actual y, después, el comienzo del siguiente.

<Slide title="El código del recorrido por niveles">
La regla **nivel por nivel** también tiene dos partes, la **visita** y el
**recorrido**; el recorrido ya no es una recursión, sino la cola: se saca el
primero, se visita, y sus hijos entran al final, uno por uno.

Un buscador de archivos que muestra primero lo más cercano recorre así: la
visita muestra cada nodo, y los nodos salen ordenados por su distancia a la
raíz.

<Slide title="El costo de un recorrido">
Los tres recorridos visitan cada nodo una sola vez. En el preorden y el
postorden, cada llamada recursiva baja por una arista, de un padre a un hijo,
y un árbol de $$n$$ nodos tiene $$n - 1$$ aristas: con la de la raíz, son
$$n$$ llamadas. En el recorrido por niveles, cada nodo entra a la cola y sale
de ella una sola vez.

| Recorrido           | Tiempo        | Memoria extra                                              |
| ------------------- | ------------- | ---------------------------------------------------------- |
| Preorden, postorden | $$\Theta(n)$$ | $$O(h)$$: una llamada pendiente por cada nodo del camino actual |
| Por niveles         | $$\Theta(n)$$ | $$O(n)$$: la cola puede guardar un nivel entero              |

La altura vuelve a aparecer: la memoria de un recorrido recursivo es la del
camino más largo desde la raíz.

{/* BinaryPicture — a static drawing of a binary tree, with the geometry of the binary-tree recipe of <SequenceStepper> (sequenceStepperTreeLayout.ts): every node in the column of its inorder position and on the row of its depth, so a figure and the widget beside it draw the same tree. File-local like HierarchyPicture (§6e-quater). A tree is { v, l, r }; values are unique within one tree, because tones, badges and notes are keyed by them, and a value may be a symbol: an expression tree keeps its operators. */}
{/* The drawings of an order, in HierarchyPicture's idiom: visits numbers the nodes in "pre" (focus), "in" (accent) or "post" (mark) order and draws an arrow from every visit to the next — along the edge when the next is a child or the parent, dashed when the walk jumps to another branch; sequence writes the visit order under the tree as a row of chips. holes draws the empty tree under every missing child, as a dashed square. ruler draws the height at the left ("h = 2"), and levelNotes writes one note per level at the right. A node toned "hidden" is not drawn, nor any edge that reaches it, but keeps its place: a tree built from the leaves up shows its subtrees loose before their parent is made; fresh names the node just added, whose edge is drawn in the mark colour. */}

{/* edgeLabels writes a word beside the edge that reaches a node, keyed by that node's value: the side it hangs on, izquierdo or derecho. frames draws a dashed box around the subtree rooted at a node, keyed by that node's value, with its name under it. */}

export function BinaryPicture({ tree, tones = {}, badges = {}, notes = {}, levels = false, strip = false, caption, size = 1, minColumns = 0, compact = false, visits, sequence = false, holes = false, ruler, levelNotes, fresh, edgeLabels = {}, frames = {}, label }) {
  const r = 17;
  // compact narrows the columns for figures set three in a row, so each fills its own column at full size.
  const pitch = compact ? 38 : 46;
  const levelH = compact ? 50 : 56;
  const pad = 16;
  const labelW = (levels ? 58 : 0) + (ruler ? 46 : 0);
  const noteW = levelNotes ? 72 : 0;
  const nodes = [];
  let column = 0;
  const walk = (t, depth, parent) => {
    if (!t) return null;
    const me = { v: t.v, depth, parent, l: null, r: null };
    me.l = walk(t.l, depth + 1, me);
    me.x = column;
    column += 1;
    nodes.push(me);
    me.r = walk(t.r, depth + 1, me);
    return me;
  };
  const root = walk(tree, 0, null);
  const depth = Math.max(0, ...nodes.map((n) => n.depth));
  const hasNotes = Object.keys(notes).length > 0;
  const holeH = holes ? 26 : 0;
  const frameH = Object.keys(frames).length > 0 ? 22 : 0;
  // strip draws the inorder under the tree, one slot per column: each value lands under its own node.
  const stripY = 2 * pad + 2 * r + depth * levelH + (hasNotes ? 12 : 0) + holeH + frameH + 6;
  const treeH = strip ? stripY + 30 + pad : stripY - 6;
  // The visit sequence: from the recursion itself.
  const seq = [];
  const order = (n) => {
    if (!n) return;
    if (visits === 'pre') seq.push(n);
    order(n.l);
    if (visits === 'in') seq.push(n);
    order(n.r);
    if (visits === 'post') seq.push(n);
  };
  if (visits) order(root);
  const rank = {};
  seq.forEach((n, i) => {
    rank[n.v] = i + 1;
  });
  const seqColor = visits === 'post' ? 'var(--color-mark)' : visits === 'in' ? 'var(--color-accent)' : 'var(--color-focus)';
  const chipW = (n) => String(n.v).length * 9 + 40;
  const gap = 20;
  const sequenceW = sequence ? seq.reduce((w, n, i) => w + chipW(n) + (i > 0 ? gap : 0), 0) : 0;
  const sequenceY = treeH + 6;
  const fullH = treeH + (sequence ? 42 : 0);
  // minColumns gives figures set side by side one canvas width, so their nodes come out the same size; the tree is centred in it.
  const columns = Math.max(column, minColumns, 1);
  const shift = ((columns - column) / 2) * pitch;
  // A caption, or the row of chips, wider than the drawing widens the canvas, and the drawing is centred in it, so nothing is clipped.
  const drawingW = labelW + 2 * pad + columns * pitch + noteW;
  const width = Math.max(drawingW, caption ? caption.length * 6.6 + 24 : 0, sequenceW + 2 * pad);
  const offset = (width - drawingW) / 2;
  const height = fullH + (caption ? 20 : 0);
  const cx = (n) => offset + labelW + pad + shift + (n.x + 0.5) * pitch;
  const cy = (n) => pad + r + n.depth * levelH;
  const rowY = (d) => pad + r + d * levelH;
  const TONE = {
    idle: ['var(--color-surface)', 'var(--color-rule-strong)', 1.3],
    focus: ['var(--color-surface)', 'var(--color-focus)', 2.4],
    mark: ['var(--color-mark-soft)', 'var(--color-mark)', 2.2],
    keep: ['var(--color-keep-soft)', 'var(--color-keep)', 2.2],
    done: ['var(--color-sunk)', 'var(--color-rule-strong)', 1.3],
    faint: ['var(--color-surface)', 'var(--color-rule)', 1.2],
  };
  const shown = nodes.filter((n) => tones[n.v] !== 'hidden');
  const head = (x, y, angle, color, key) => (
    <polygon key={key} points="0,0 -8,-4.5 -8,4.5" transform={`translate(${x} ${y}) rotate(${angle})`} style={{ fill: color }} />
  );
  const dot = (x, y, k, color, key) => (
    <g key={key}>
      <circle cx={x} cy={y} r="8.5" style={{ fill: color }} />
      <text x={x} y={y + 3.8} textAnchor="middle" fontSize="10.5" fontWeight="700" style={{ fill: 'var(--color-ground)' }}>{k}</text>
    </g>
  );
  // An edge between a parent and a child, trimmed to the two circles.
  const ends = (p, c) => {
    const dx = cx(c) - cx(p);
    const dy = cy(c) - cy(p);
    const len = Math.hypot(dx, dy);
    return { x1: cx(p) + (dx / len) * r, y1: cy(p) + (dy / len) * r, x2: cx(c) - (dx / len) * r, y2: cy(c) - (dy / len) * r, angle: (Math.atan2(dy, dx) * 180) / Math.PI };
  };
  // From one visit to the next: along the edge, down to a child or up to the parent; otherwise a dashed jump that leaves the right of one and enters the left of the next, both ends level, so the head points the way the line arrives.
  const step = (a, b, key) => {
    const down = b.parent === a;
    const up = a.parent === b;
    if (down || up) {
      const e = down ? ends(a, b) : ends(b, a);
      const mx = (e.x1 + e.x2) / 2;
      const my = (e.y1 + e.y2) / 2;
      const rad = (e.angle * Math.PI) / 180;
      const s = down ? 4 : -4;
      return (
        <g key={key}>
          <line x1={e.x1} y1={e.y1} x2={e.x2} y2={e.y2} style={{ stroke: seqColor }} strokeWidth="2.4" />
          {head(mx + s * Math.cos(rad), my + s * Math.sin(rad), down ? e.angle : e.angle + 180, seqColor, `${key}-h`)}
        </g>
      );
    }
    const sx = cx(a) + r + 2;
    const sy = cy(a) + 2;
    const ex = cx(b) - r - 2;
    const ey = cy(b) + 2;
    const d = Math.max(18, (ex - sx) / 2);
    return (
      <g key={key}>
        <path d={`M ${sx} ${sy} C ${sx + d} ${sy}, ${ex - d} ${ey}, ${ex - 7} ${ey}`} fill="none" style={{ stroke: seqColor }} strokeWidth="1.8" strokeDasharray="5 4" />
        {head(ex, ey, 0, seqColor, `${key}-h`)}
      </g>
    );
  };
  return (
    <svg viewBox={`0 0 ${width} ${height}`} style={{ width: '100%', maxWidth: `${Math.round(width * 1.4 * size)}px`, height: 'auto', display: 'block', margin: '1rem auto' }} role="img" aria-label={label}>
      {levels
        ? [...Array(depth + 1).keys()].map((d) => (
            <g key={`lv${d}`}>
              <line x1={offset + labelW - 4} y1={rowY(d)} x2={offset + drawingW - noteW - 4} y2={rowY(d)} style={{ stroke: 'var(--color-rule)' }} strokeWidth="1" strokeDasharray="3 5" />
              <text x={offset + (ruler ? 50 : 4)} y={rowY(d) + 4} fontSize="11" style={{ fill: 'var(--color-ink-faint)' }}>{`nivel ${d}`}</text>
            </g>
          ))
        : null}
      {ruler ? (
        <g>
          <line x1={offset + 24} y1={rowY(0)} x2={offset + 24} y2={rowY(depth)} style={{ stroke: 'var(--color-ink)' }} strokeWidth="1.8" />
          {[...Array(depth + 1).keys()].map((d) => (
            <line key={`rt${d}`} x1={offset + 18} y1={rowY(d)} x2={offset + 30} y2={rowY(d)} style={{ stroke: 'var(--color-ink)' }} strokeWidth="1.8" />
          ))}
          <text x={offset + 12} y={(rowY(0) + rowY(depth)) / 2} textAnchor="middle" fontSize="13" fontWeight="700" transform={`rotate(-90 ${offset + 12} ${(rowY(0) + rowY(depth)) / 2})`} style={{ fill: 'var(--color-ink)' }}>{ruler}</text>
        </g>
      ) : null}
      {levelNotes
        ? levelNotes.map((note, d) => (
            <text key={`ln${d}`} x={offset + drawingW - noteW + 8} y={rowY(d) + 4} fontSize="12" fontWeight="600" style={{ fill: 'var(--color-ink-soft)' }}>{note}</text>
          ))
        : null}
      {Object.entries(frames).map(([key, text]) => {
        const top = nodes.find((n) => String(n.v) === key);
        if (!top) return null;
        const subtree = (n) => (n ? [n, ...subtree(n.l), ...subtree(n.r)] : []);
        const members = subtree(top);
        const x1 = Math.min(...members.map(cx)) - r - 8;
        const x2 = Math.max(...members.map(cx)) + r + 8;
        const y1 = cy(top) - r - 7;
        const y2 = Math.max(...members.map(cy)) + r + 7;
        return (
          <g key={`f${key}`}>
            <rect x={x1} y={y1} width={x2 - x1} height={y2 - y1} rx="12" style={{ fill: 'var(--color-keep-soft)', stroke: 'var(--color-keep)' }} strokeWidth="1.4" strokeDasharray="5 4" />
            <text x={(x1 + x2) / 2} y={y2 + 15} textAnchor="middle" fontSize="12" fontWeight="700" style={{ fill: 'var(--color-ink-soft)' }}>{text}</text>
          </g>
        );
      })}
      {shown.map((n) => {
        if (n.parent === null || tones[n.parent.v] === 'hidden') return null;
        const e = ends(n.parent, n);
        const faint = tones[n.v] === 'faint';
        const isFresh = n.v === fresh;
        return <line key={`e${n.v}`} x1={e.x1} y1={e.y1} x2={e.x2} y2={e.y2} style={{ stroke: isFresh ? 'var(--color-mark)' : faint ? 'var(--color-rule)' : 'var(--color-rule-strong)' }} strokeWidth={isFresh ? 2.6 : 1.3} strokeDasharray={faint ? '4 3' : undefined} />;
      })}
      {shown
        .filter((n) => n.parent !== null && edgeLabels[n.v] !== undefined)
        .map((n) => {
          const e = ends(n.parent, n);
          const left = n === n.parent.l;
          return (
            <text key={`el${n.v}`} x={(e.x1 + e.x2) / 2 + (left ? -8 : 8)} y={(e.y1 + e.y2) / 2 + 3} textAnchor={left ? 'end' : 'start'} fontSize="12" fontWeight="700" paintOrder="stroke" strokeWidth="3" style={{ fill: 'var(--color-ink-soft)', stroke: 'var(--color-ground)' }}>{edgeLabels[n.v]}</text>
          );
        })}
      {holes
        ? shown.flatMap((n) =>
            ['l', 'r']
              .filter((side) => n[side] === null || tones[n[side].v] === 'hidden')
              .map((side) => {
                const x = cx(n) + (side === 'l' ? -1 : 1) * pitch * 0.32;
                const y = cy(n) + r + 14;
                return (
                  <g key={`h${n.v}${side}`}>
                    <line x1={cx(n) + (side === 'l' ? -1 : 1) * r * 0.6} y1={cy(n) + r * 0.8} x2={x} y2={y - 4} style={{ stroke: 'var(--color-rule)' }} strokeWidth="1.1" strokeDasharray="2 3" />
                    <rect x={x - 6} y={y - 4} width="12" height="10" rx="2" style={{ fill: 'none', stroke: 'var(--color-rule-strong)' }} strokeWidth="1" strokeDasharray="2 2" />
                  </g>
                );
              }),
          )
        : null}
      {visits ? seq.slice(1).map((b, i) => step(seq[i], b, `s${b.v}`)) : null}
      {shown.length === 0 && root ? (
        <text x={cx(root)} y={cy(root) + 5} textAnchor="middle" fontSize="13" fontFamily="var(--font-mono, monospace)" style={{ fill: 'var(--color-ink-soft)' }}>root = null</text>
      ) : null}
      {shown.map((n) => {
        const [fill, stroke, weight] = TONE[tones[n.v] ?? 'idle'];
        const faint = tones[n.v] === 'faint';
        return (
          <g key={`n${n.v}`} opacity={faint ? 0.5 : undefined}>
            <circle cx={cx(n)} cy={cy(n)} r={r} style={{ fill, stroke }} strokeWidth={weight} strokeDasharray={faint ? '4 3' : undefined} />
            <text x={cx(n)} y={cy(n) + 5} textAnchor="middle" fontSize="14" fontWeight="600" style={{ fill: 'var(--color-ink)' }}>{n.v}</text>
            {badges[n.v] !== undefined ? (
              <text x={cx(n) + r - 1} y={cy(n) - r + 3} fontSize="10.5" fontWeight="700" paintOrder="stroke" strokeWidth="3" style={{ fill: 'var(--color-focus)', stroke: 'var(--color-ground)' }}>{badges[n.v]}</text>
            ) : null}
            {notes[n.v] !== undefined ? (
              <text x={cx(n)} y={cy(n) + r + 12} textAnchor="middle" fontSize="10" paintOrder="stroke" strokeWidth="3" style={{ fill: 'var(--color-ink-soft)', stroke: 'var(--color-ground)' }}>{notes[n.v]}</text>
            ) : null}
          </g>
        );
      })}
      {visits ? seq.map((n) => dot(cx(n) - r * 0.8, cy(n) - r * 0.8, rank[n.v], seqColor, `d${n.v}`)) : null}
      {strip
        ? nodes.map((n) => (
            <g key={`s${n.v}`}>
              <line x1={cx(n)} y1={cy(n) + r + 3} x2={cx(n)} y2={stripY - 3} style={{ stroke: 'var(--color-rule)' }} strokeWidth="1" strokeDasharray="2 4" />
              <rect x={cx(n) - 19} y={stripY} width="38" height="30" rx="3" style={{ fill: 'var(--color-surface)', stroke: visits === 'in' ? seqColor : 'var(--color-rule-strong)' }} strokeWidth={visits === 'in' ? 1.6 : 1} />
              <text x={cx(n)} y={stripY + 20} textAnchor="middle" fontSize="14" fontWeight="600" style={{ fill: 'var(--color-ink)' }}>{n.v}</text>
            </g>
          ))
        : null}
      {sequence
        ? (() => {
            let x = (width - sequenceW) / 2;
            return seq.map((n, i) => {
              const w = chipW(n);
              const x0 = x + (i > 0 ? gap : 0);
              x = x0 + w;
              return (
                <g key={`c${n.v}`}>
                  {i > 0 ? <line x1={x0 - gap + 3} y1={sequenceY + 14} x2={x0 - 6} y2={sequenceY + 14} style={{ stroke: seqColor }} strokeWidth="1.6" /> : null}
                  {i > 0 ? head(x0 - 2, sequenceY + 14, 0, seqColor, `ch${n.v}`) : null}
                  <rect x={x0} y={sequenceY} width={w} height="28" rx="14" style={{ fill: 'var(--color-surface)', stroke: seqColor }} strokeWidth="1.6" />
                  {dot(x0 + 13, sequenceY + 14, rank[n.v], seqColor, `cd${n.v}`)}
                  <text x={x0 + 26 + (w - 26) / 2} y={sequenceY + 19} textAnchor="middle" fontSize="14" fontWeight="600" style={{ fill: 'var(--color-ink)' }}>{n.v}</text>
                </g>
              );
            });
          })()
        : null}
      {caption ? (
        <text x={width / 2} y={fullH + 12} textAnchor="middle" fontSize="12" style={{ fill: 'var(--color-ink-soft)' }}>{caption}</text>
      ) : null}
    </svg>
  );
}

{/* BinaryMemory — a binary tree as it sits in memory, in TreeMemory's idiom: a node is a box with its value between two grey compartments, left and right, whose dots are the references to its two children; a missing child is a reference to null. Nodes sit in the columns of their inorder position, as BinaryPicture places them. entry draws the BinaryTree object and its root reference, which points to null when the tree is empty; fields names the three fields over the root. File-local like HierarchyPicture (§6e-quater). */}

export function BinaryMemory({ tree, entry = false, fields = false, px, caption, label }) {
  const H = 28;
  const COMP = 18;
  const pad = 14;
  const top = pad + (fields ? 30 : 0);
  const vw = (v) => Math.max(30, String(v).length * 9 + 16);
  const nodes = [];
  let column = 0;
  const walk = (t, depth, parent) => {
    if (!t) return null;
    const me = { v: t.v, depth, parent, l: null, r: null };
    me.l = walk(t.l, depth + 1, me);
    me.x = column;
    column += 1;
    nodes.push(me);
    me.r = walk(t.r, depth + 1, me);
    return me;
  };
  const root = walk(tree, 0, null);
  const depth = Math.max(0, ...nodes.map((n) => n.depth));
  const pitch = nodes.length > 0 ? Math.max(...nodes.map((n) => vw(n.v))) + 2 * COMP + 12 : 40;
  const LEVEL = 92;
  const entryW = entry ? 92 : 0;
  const drawingW = entryW + 2 * pad + Math.max(column, 1) * pitch;
  const width = Math.max(drawingW, caption ? caption.length * 6.6 + 24 : 0);
  const offset = (width - drawingW) / 2;
  // The room under the leaves is for their null references; an empty tree has none.
  const bodyH = top + depth * LEVEL + H + (nodes.length > 0 ? 40 : 8) + pad;
  const height = bodyH + (caption ? 18 : 0);
  const cx = (n) => offset + entryW + pad + (n.x + 0.5) * pitch;
  const y = (n) => top + n.depth * LEVEL;
  const boxW = (n) => vw(n.v) + 2 * COMP;
  const arrow = (x1, y1, x2, y2, key, color = 'var(--color-ink-soft)') => {
    const a = Math.atan2(y2 - y1, x2 - x1);
    const h = 7;
    const tip = (s) => `${x2 - h * Math.cos(a + s)},${y2 - h * Math.sin(a + s)}`;
    return (
      <g key={key}>
        <line x1={x1} y1={y1} x2={x2 - 5 * Math.cos(a)} y2={y2 - 5 * Math.sin(a)} style={{ stroke: color }} strokeWidth="1.6" />
        <path d={`M ${x2},${y2} L ${tip(0.42)} L ${tip(-0.42)} z`} style={{ fill: color }} />
      </g>
    );
  };
  const mono = 'var(--font-mono, monospace)';
  return (
    <svg viewBox={`0 0 ${width} ${height}`} style={{ width: px ? `${Math.round(width * px)}px` : '100%', maxWidth: px ? '100%' : `${Math.round(width * 1.3)}px`, height: 'auto', display: 'block', margin: '0.75rem auto' }} role="img" aria-label={label}>
      {entry ? (
        <g>
          <text x={offset + pad} y={top - 6} fontSize="11" style={{ fill: 'var(--color-ink-faint)' }}>BinaryTree</text>
          <rect x={offset + pad} y={top} width="52" height={H} rx="4" style={{ fill: 'var(--color-surface)', stroke: 'var(--color-rule-strong)' }} strokeWidth="1.6" />
          <text x={offset + pad + 26} y={top + 18} textAnchor="middle" fontSize="12" fontFamily={mono} style={{ fill: 'var(--color-ink)' }}>root</text>
          <rect x={offset + pad + 52} y={top} width={COMP} height={H} rx="3" style={{ fill: 'var(--color-sunk)', stroke: 'var(--color-rule-strong)' }} strokeWidth="1.6" />
          <circle cx={offset + pad + 52 + COMP / 2} cy={top + H / 2} r="2.6" style={{ fill: 'var(--color-ink-soft)' }} />
          {root ? (
            arrow(offset + pad + 52 + COMP / 2, top + H / 2, cx(root) - boxW(root) / 2 - 2, top + H / 2, 'entry', 'var(--color-focus)')
          ) : (
            <g>
              {arrow(offset + pad + 52 + COMP / 2, top + H / 2, offset + pad + 52 + COMP + 30, top + H / 2, 'entry', 'var(--color-rule-strong)')}
              <text x={offset + pad + 52 + COMP + 34} y={top + H / 2 + 4} fontSize="11" fontFamily={mono} style={{ fill: 'var(--color-ink-faint)' }}>null</text>
            </g>
          )}
        </g>
      ) : null}
      {nodes.map((n) => {
        const w = boxW(n);
        const x = cx(n) - w / 2;
        const lx = x + COMP / 2;
        const rx = x + w - COMP / 2;
        const ny = y(n);
        return (
          <g key={`m${n.v}`}>
            {fields && n.parent === null ? (
              <g>
                {/* The three names would overlap over a narrow box: value above it, left and right out to its sides, each pointing at its field. */}
                <text x={cx(n)} y={ny - 18} textAnchor="middle" fontSize="10" fontFamily={mono} style={{ fill: 'var(--color-ink-faint)' }}>value</text>
                <line x1={cx(n)} y1={ny - 14} x2={cx(n)} y2={ny - 2} style={{ stroke: 'var(--color-rule)' }} strokeWidth="1" />
                <text x={lx - 14} y={ny - 6} textAnchor="end" fontSize="10" fontFamily={mono} style={{ fill: 'var(--color-ink-faint)' }}>left</text>
                <line x1={lx - 12} y1={ny - 9} x2={lx - 2} y2={ny + 2} style={{ stroke: 'var(--color-rule)' }} strokeWidth="1" />
                <text x={rx + 14} y={ny - 6} textAnchor="start" fontSize="10" fontFamily={mono} style={{ fill: 'var(--color-ink-faint)' }}>right</text>
                <line x1={rx + 12} y1={ny - 9} x2={rx + 2} y2={ny + 2} style={{ stroke: 'var(--color-rule)' }} strokeWidth="1" />
              </g>
            ) : null}
            <rect x={x} y={ny} width={COMP} height={H} rx="3" style={{ fill: 'var(--color-sunk)', stroke: 'var(--color-rule-strong)' }} strokeWidth="1.6" />
            <rect x={x + COMP} y={ny} width={w - 2 * COMP} height={H} rx="3" style={{ fill: 'var(--color-surface)', stroke: 'var(--color-rule-strong)' }} strokeWidth="1.6" />
            <rect x={x + w - COMP} y={ny} width={COMP} height={H} rx="3" style={{ fill: 'var(--color-sunk)', stroke: 'var(--color-rule-strong)' }} strokeWidth="1.6" />
            <text x={cx(n)} y={ny + 19} textAnchor="middle" fontSize="14" fontWeight="600" style={{ fill: 'var(--color-ink)' }}>{n.v}</text>
            {[['l', lx], ['r', rx]].map(([side, sx]) => {
              const child = n[side];
              return (
                <g key={`${n.v}${side}`}>
                  <circle cx={sx} cy={ny + H / 2} r="2.6" style={{ fill: 'var(--color-ink-soft)' }} />
                  {child
                    ? arrow(sx, ny + H / 2, cx(child), y(child) - 1, `a${n.v}${side}`)
                    : (
                      <g>
                        {arrow(sx, ny + H / 2, sx + (side === 'l' ? -6 : 6), ny + H + 18, `z${n.v}${side}`, 'var(--color-rule-strong)')}
                        <text x={sx + (side === 'l' ? -6 : 6)} y={ny + H + 30} textAnchor="middle" fontSize="10" fontFamily={mono} style={{ fill: 'var(--color-ink-faint)' }}>null</text>
                      </g>
                    )}
                </g>
              );
            })}
          </g>
        );
      })}
      {caption ? (
        <text x={width / 2} y={bodyH + 6} textAnchor="middle" fontSize="12" style={{ fill: 'var(--color-ink-soft)' }}>{caption}</text>
      ) : null}
    </svg>
  );
}

{/* NodeGraph — nodes at explicit grid positions, joined by edges: for a shape no tree layout can draw, such as a node with two parents. Nodes are BinaryPicture's circles; a node or an edge toned "mark" is the one that breaks the rule the drawing shows. File-local like HierarchyPicture (§6e-quater). */}

export function NodeGraph({ nodes, edges, cols, rows, tones = {}, markEdges = [], caption, label }) {
  const r = 17;
  const pitch = 52;
  const levelH = 58;
  const pad = 18;
  const drawingW = 2 * pad + cols * pitch;
  const width = Math.max(drawingW, caption ? caption.length * 6.6 + 24 : 0);
  const offset = (width - drawingW) / 2;
  const treeH = 2 * pad + 2 * r + (rows - 1) * levelH;
  const height = treeH + (caption ? 20 : 0);
  const at = Object.fromEntries(nodes.map((n) => [n.v, { x: offset + pad + (n.x + 0.5) * pitch, y: pad + r + n.y * levelH }]));
  const marked = new Set(markEdges.map(([a, b]) => `${a}>${b}`));
  return (
    <svg viewBox={`0 0 ${width} ${height}`} style={{ width: '100%', maxWidth: `${Math.round(width * 1.4)}px`, height: 'auto', display: 'block', margin: '1rem auto' }} role="img" aria-label={label}>
      {edges.map(([a, b]) => {
        const p = at[a];
        const c = at[b];
        const dx = c.x - p.x;
        const dy = c.y - p.y;
        const len = Math.hypot(dx, dy);
        const hot = marked.has(`${a}>${b}`);
        return <line key={`${a}>${b}`} x1={p.x + (dx / len) * r} y1={p.y + (dy / len) * r} x2={c.x - (dx / len) * r} y2={c.y - (dy / len) * r} style={{ stroke: hot ? 'var(--color-mark)' : 'var(--color-rule-strong)' }} strokeWidth={hot ? 2.6 : 1.3} />;
      })}
      {nodes.map((n) => {
        const hot = tones[n.v] === 'mark';
        return (
          <g key={`n${n.v}`}>
            <circle cx={at[n.v].x} cy={at[n.v].y} r={r} style={{ fill: hot ? 'var(--color-mark-soft)' : 'var(--color-surface)', stroke: hot ? 'var(--color-mark)' : 'var(--color-rule-strong)' }} strokeWidth={hot ? 2.2 : 1.3} />
            <text x={at[n.v].x} y={at[n.v].y + 5} textAnchor="middle" fontSize="14" fontWeight="600" style={{ fill: 'var(--color-ink)' }}>{n.v}</text>
          </g>
        );
      })}
      {caption ? (
        <text x={width / 2} y={treeH + 12} textAnchor="middle" fontSize="12" style={{ fill: 'var(--color-ink-soft)' }}>{caption}</text>
      ) : null}
    </svg>
  );
}

<Slide title="Jerarquías de a dos">

<Slide title="Dos hijos con nombre">
La clase árbol ya puede guardar estos árboles: los hijos de cada nodo van en
su secuencia, y el primero hace de izquierdo. Pero la secuencia no sabe lo
que estos problemas garantizan: admite un tercer hijo, y el papel de cada uno
queda en su posición, primero o segundo, y no en un nombre.

Cuando en un problema ningún nodo tiene más de dos hijos, y cada uno tiene su
papel, conviene que la estructura lo diga. Un **árbol binario** es un árbol
en el que cada nodo tiene **a lo más dos hijos**, y cada hijo tiene un
nombre: es el **izquierdo** o el **derecho**.

</div>
<div style={{ flex: '2 1 15rem', display: 'grid', gridTemplateColumns: '1fr 1fr', gap: '1rem', alignItems: 'end' }}>
<BinaryPicture tree={{ v: 7, l: { v: 3 } }} minColumns={3} edgeLabels={{ 3: 'izquierdo' }} label="Un nodo 7 con un solo hijo, el 3, a la izquierda." caption="solo un hijo izquierdo" />
<BinaryPicture tree={{ v: 7, r: { v: 3 } }} minColumns={3} edgeLabels={{ 3: 'derecho' }} label="Un nodo 7 con un solo hijo, el 3, a la derecha." caption="solo un hijo derecho" />

Cuando un nodo tiene un solo hijo, ese hijo conserva su lado: un nodo con
solo un hijo izquierdo y uno con solo un hijo derecho son dos árboles
binarios distintos. En una secuencia serían el mismo: un nodo con un solo
hijo, el primero.

</Slide>

<Slide title="Una definición recursiva">
Como en el árbol general, cada hijo es la raíz de un subárbol. En un árbol
binario, cada nodo tiene dos lugares para ellos: el **subárbol izquierdo** y
el **subárbol derecho**.

Si a un nodo le falta un hijo, en ese lugar queda un subárbol sin nodos: el
**árbol vacío**. Una hoja tiene sus dos subárboles vacíos.

La definición, entonces, usa árboles binarios más pequeños, como la del árbol
general: es **recursiva**. Un árbol binario **es el árbol vacío, o es un nodo
con un valor y dos árboles binarios debajo**, el izquierdo y el derecho.

</Slide>

<Slide title="Las invariantes del árbol binario">
Un árbol binario sostiene dos invariantes en todo momento:

1. **Cada nodo tiene a lo más dos hijos, y cada uno es el izquierdo o el
   derecho.** Es lo que permite escribir el nodo con dos campos fijos.
2. **Cada nodo, salvo la raíz, tiene exactamente un padre.** Por eso hay un
   único camino desde la raíz hasta cada nodo, y un recorrido visita cada
   uno una sola vez.

Ninguna de las dos dice nada sobre los **valores**: un árbol binario no
ordena lo que guarda.

</Slide>

<Slide title="El nodo binario">
Un nodo binario guarda tres cosas: su **valor** y una referencia a cada hijo,
`left` al izquierdo y `right` al derecho. Son dos campos con nombre, en vez de
la secuencia de hijos del árbol general.

<PresentationWide fraction={0.75}>

<SideBySide direction="mode-aware" left="El código" right="En memoria" align="center">

```java
class Node<T> {
    T value;
    Node<T> left;
    Node<T> right;

    Node(T value) {
        this.value = value;
    }

    Node(T value, Node<T> left, Node<T> right) {
        this.value = value;
        this.left = left;
        this.right = right;
    }
}
```

<BinaryMemory tree={{ v: '+', l: { v: 3 }, r: { v: 4 } }} fields label="Un nodo en memoria: el +, con una referencia a su hijo izquierdo, el 3, y otra a su hijo derecho, el 4. Las referencias de las hojas apuntan a null." caption="el + apunta a sus dos hijos; el 3 y el 4, a null" />

</SideBySide>

</PresentationWide>

Un hijo que falta es una referencia en `null`: en ese lado está el árbol
vacío.

Hay dos constructores. El primero recibe solo el valor: el nodo nace con
`left` y `right` en `null`, como una hoja, y sus hijos se cuelgan después. El
segundo recibe también sus dos hijos, ya armados. Cuál conviene depende de
cómo se arme el árbol, y eso depende del problema: cada implementación trae
los constructores que su uso necesita.

</Slide>

<Slide title="El árbol binario y su raíz">
El árbol binario guarda una sola referencia: `root`, a su raíz. Como en el
árbol general, es la entrada: desde ella, siguiendo `left` y `right`, se
llega a cualquier nodo.

```java
class BinaryTree<T> {
    Node<T> root;
}
```

<Slide title="El preorden en un árbol binario">
En **preorden**, cada nodo se visita antes que sus dos subárboles: primero el
izquierdo, después el derecho.

<OrderRule parts={['raíz', 'izquierda', 'derecha']} label="La regla del preorden: la raíz, después el subárbol izquierdo, después el derecho." />

En una expresión, el preorden escribe la **notación prefija**: cada operador
antes que sus operandos.

<BinaryPicture tree={EXPR} visits="pre" sequence label="La expresión en preorden: cada nodo numerado en el orden en que se visita, con flechas de cada visita a la siguiente. </Slide>

<Slide title="El código del preorden en un árbol binario">
La regla se escribe con las mismas dos partes, la **visita** y el
**recorrido**, más una línea nueva: el **caso base**. Un subárbol vacío es
`null`, y en él no hay nada que visitar.

<Slide title="El inorden">
El **inorden** visita cada nodo **entre** sus dos subárboles: primero el
izquierdo, después el nodo, después el derecho. Solo existe cuando los dos
hijos tienen nombre.

<OrderRule parts={['izquierda', 'raíz', 'derecha']} tone="accent" label="La regla del inorden: el subárbol izquierdo, después la raíz, después el derecho." />

Cada nodo está dibujado en la columna de su lugar en el inorden: el inorden se
lee de izquierda a derecha, y cada valor cae justo debajo de su nodo. En una
expresión, el inorden la escribe como se escribe a mano, pero **sin los
paréntesis**: la agrupación la guarda el árbol, no el texto.

<BinaryPicture tree={EXPR} visits="in" strip label="La expresión en inorden: cada nodo numerado en el orden en que se visita, con flechas de cada visita a la siguiente, y debajo de cada nodo su valor. La fila de abajo dice 3 + 4 * 5 − 1." caption="3 + 4 * 5 − 1: sin paréntesis se leería 22; el árbol vale 28" />

</Slide>

<Slide title="El código del inorden">
La visita va entre las dos llamadas, así que el recorrido queda partido en
dos: primero la izquierda, y después de la visita, la derecha.

<SequenceStepper eda="binary-tree" operation="inorder" values={['*', '+', '−', 3, 4, 5, 1]} />

</Slide>

<Slide title="El postorden en un árbol binario">
En **postorden**, cada nodo se visita después de sus dos subárboles: primero
el izquierdo, después el derecho, y al final el nodo.

<OrderRule parts={['izquierda', 'derecha', 'raíz']} tone="mark" label="La regla del postorden: el subárbol izquierdo, después el derecho, después la raíz." />

En una expresión, el postorden escribe la **notación polaca inversa**: cada
operador después de sus operandos, la que se evalúa con una pila.

<BinaryPicture tree={EXPR} visits="post" sequence label="La expresión en postorden: cada nodo numerado en el orden en que se visita, con flechas de cada visita a la siguiente. 3 primero, 4 segundo, más 3, 5 cuarto, 1 quinto, menos 6, asterisco 7. Debajo, la misma secuencia: la notación polaca inversa." caption="notación polaca inversa: cada operador después de sus operandos" />

</Slide>

<Slide title="El código del postorden en un árbol binario">
La visita va al final: un nodo se visita cuando sus dos subárboles ya se
recorrieron.

<SequenceStepper eda="binary-tree" operation="postorder" values={['*', '+', '−', 3, 4, 5, 1]} />

</Slide>

<Slide title="La altura por recursión">
La altura sigue la definición recursiva al pie de la letra. El árbol vacío
tiene altura $$-1$$, para que una hoja, con dos subárboles vacíos, tenga
altura $$1 + \max(-1, -1) = 0$$. Un nodo tiene una unidad más que el más alto
de sus dos subárboles, así que responde **después** de ellos: es un
postorden.

<Slide title="Estricto, lleno y completo">
Tres formas de árbol binario tienen nombre:

- **Estricto**: cada nodo tiene cero o dos hijos, nunca uno.
- **Lleno**: es estricto y todas sus hojas están en el último nivel; cada
  nivel tiene todos los nodos que caben.
- **Completo**: todos los niveles están llenos, salvo quizás el último, que
  se llena de izquierda a derecha, sin huecos.

Un árbol de expresión es siempre estricto: cada operador tiene exactamente dos
operandos.

<Slide title="Cuántos nodos caben en una altura">
Cada nivel de un árbol binario tiene, a lo más, el doble de nodos que el
anterior: el nivel $$d$$ tiene a lo más $$2^d$$. Un árbol **lleno** de altura
$$h$$ los tiene todos:

$$
1 + 2 + 4 + \cdots + 2^h = 2^{h+1} - 1
$$

<BinaryPicture tree={EXPR} levels ruler="h = 2" levelNotes={['1 nodo', '2 nodos', '4 nodos']} label="El árbol de la expresión, lleno, con una línea por nivel y la altura a la izquierda, h = 2: un nodo en el nivel 0, dos en el nivel 1, cuatro en el nivel 2; siete en total." caption="altura 2: 1 + 2 + 4 = 7 nodos" />

Ningún árbol binario de altura $$h$$ tiene más nodos que el lleno.

</Slide>

<Slide title="La altura mínima y la máxima">
Con $$n$$ nodos, la altura de un árbol binario queda entre dos extremos:

- **La más baja posible es $$\lfloor \log_2 n \rfloor$$**: los nodos llenan
  cada nivel antes de abrir el siguiente, como en un árbol completo. Como un
  árbol de altura $$h$$ tiene a lo más $$2^{h+1} - 1$$ nodos, hacen falta
  al menos $$\lfloor \log_2 n \rfloor$$ niveles bajo la raíz.
- **La más alta posible es $$n - 1$$**: cada nodo tiene un solo hijo, y el
  árbol es una cadena, una lista enlazada dibujada en diagonal.

<BinaryPicture tree={{ v: 4, l: { v: 2, l: { v: 1 }, r: { v: 3 } }, r: { v: 6, l: { v: 5 }, r: { v: 7 } } }} ruler="h = 2" label="Siete nodos en un árbol de altura 2, la mínima, con la altura marcada a la izquierda." caption="7 nodos, altura 2" />
<BinaryPicture tree={{ v: 1, r: { v: 2, r: { v: 3, r: { v: 4, r: { v: 5, r: { v: 6, r: { v: 7 } } } } } } }} size={0.75} ruler="h = 6" label="Los mismos siete nodos en una cadena: cada nodo tiene un solo hijo, a la derecha. Altura 6, la máxima, marcada a la izquierda." caption="7 nodos, altura 6" />
</div>

Todas las operaciones que bajan por el árbol pagan su altura, y entre esos
dos extremos la diferencia es la que hay entre $$\log_2 n$$ y $$n$$.

</Slide>

<Slide title="Costos del árbol binario">
Cada operación recorre el árbol entero, porque sin un orden entre los
valores no hay subárbol que pueda saltarse:

| Molde     | Operación                         | Peor caso     | Invariante que restaura                  |
| --------- | --------------------------------- | ------------- | ---------------------------------------- |
| Consultar | preorden, inorden, postorden      | $$\Theta(n)$$ | Ninguna: no cambian el árbol             |
| Consultar | altura                            | $$\Theta(n)$$ | Ninguna: no cambia el árbol              |
| Modificar | espejo                            | $$\Theta(n)$$ | Ninguna: el intercambio mantiene las dos |

</Slide>

<Slide title="Buscar en un árbol binario">
Para saber si un valor está en un árbol binario, la única regla posible es
mirar el nodo y, si no es, buscar en sus **dos** subárboles: nada dice en cuál
de los dos puede estar.

<SequenceStepper eda="binary-tree" operation="search" values={['*', '+', '−', 3, 4, 5, 1]} target={5} />

Cuando el valor no está, la búsqueda compara los $$n$$ nodos: cuesta
$$\Theta(n)$$, lo mismo que buscar en una lista. La jerarquía organiza los
datos, pero sola no dice dónde buscar.

</Slide>

## El árbol binario de búsqueda

<Slide title="La búsqueda binaria en un arreglo ordenado">
En un arreglo **ordenado** se busca sin recorrerlo: se compara con el
elemento del medio y, según sea mayor o menor, se descarta la mitad que no
puede contenerlo. Cada comparación parte el problema en dos, así que bastan
$$\lfloor \log_2 n \rfloor + 1$$ comparaciones.

<SortedRow values={[1, 2, 3, 4, 5, 6, 7, 8, 9]} hot={4} out={[0, 1, 2, 3]} label="El arreglo ordenado del 1 al 9. Buscando el 7, la primera comparación es con el 5, el del medio, y la mitad izquierda, del 1 al 4, queda descartada." caption="buscar 7: se compara con 5, el del medio, y la mitad izquierda se descarta" />

El problema es **insertar**. Para que el arreglo siga ordenado, un valor
nuevo tiene que entrar en su lugar, y todos los mayores se corren una
casilla: $$\Theta(n)$$.

</Slide>

<Slide title="Las decisiones de la búsqueda binaria">
Las comparaciones de una búsqueda binaria siempre siguen el mismo patrón. La
primera es con el elemento del medio; si el buscado es menor, la segunda es
con el medio de la mitad izquierda; si es mayor, con el medio de la derecha;
y así. Dibujadas, esas decisiones forman un **árbol binario**: el medio es la
raíz, y cada mitad, un subárbol.

<BinaryPicture tree={FROM_ARRAY} strip label="El árbol de decisiones de una búsqueda binaria sobre el arreglo del 1 al 9. La raíz es 5; a la izquierda, 2, con hijos 1 y 3, y el 3 con un hijo derecho, 4; a la derecha, 7, con hijos 6 y 8, y el 8 con un hijo derecho, 9. Debajo, el arreglo ordenado, con cada valor exactamente bajo su nodo." caption="el arreglo ordenado es el inorden del árbol" />

Si, en lugar del arreglo, se guardan esas decisiones como **nodos
enlazados**, buscar sigue siendo comparar y bajar por un lado. E insertar ya
no corre nada: el valor nuevo se cuelga como una hoja en el lugar donde su
búsqueda termina.

</Slide>

<Slide title="La invariante de orden">
Un **árbol binario de búsqueda** —en inglés, *binary search tree*, o
**BST**— es un árbol binario que cumple esta **invariante**:

> **para cada nodo, todos los valores de su subárbol izquierdo son menores
> que el suyo, y todos los de su subárbol derecho son mayores.**

La palabra que importa es **todos**. No basta con que cada hijo esté del lado
correcto de su padre: un valor tiene que estar del lado correcto de **cada**
nodo por encima de él.

<div style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fit, minmax(220px, 1fr))', gap: '1rem', alignItems: 'end' }}>
<BinaryPicture tree={{ v: 5, l: { v: 3, l: { v: 1 }, r: { v: 4 } }, r: { v: 8 } }} minColumns={5} label="Un árbol de búsqueda: 5 con hijos 3 y 8; el 3 con hijos 1 y 4. Todo lo que está a la izquierda del 5 es menor que 5, y todo lo de la derecha, mayor." caption="cumple la invariante en cada nodo" />
<BinaryPicture tree={{ v: 5, l: { v: 3, l: { v: 1 }, r: { v: 7 } }, r: { v: 8 } }} tones={{ 7: 'mark' }} notes={{ 7: '7 > 5' }} minColumns={5} label="Un árbol que no es de búsqueda: 5 con hijos 3 y 8; el 3 con hijos 1 y 7. Cada hijo está del lado correcto de su padre, pero el 7 está en el subárbol izquierdo del 5 y es mayor que 5." caption="7 está bien con su padre, 3, pero no con el 5" />
</div>

</Slide>

<Slide title="Las invariantes del árbol de búsqueda">
Un árbol binario de búsqueda sostiene tres propiedades:

1. **Cada nodo tiene a lo más dos hijos, izquierdo y derecho.**
2. **Cada nodo, salvo la raíz, tiene exactamente un padre.**
3. **Todo el subárbol izquierdo de un nodo es menor que él, y todo el derecho
   es mayor.** En particular, ningún valor se repite.

Las dos primeras son las del árbol binario, sin cambios. La tercera es la
nueva, y la que decide los costos: es la que permite descartar un subárbol
entero en cada comparación, y la que cada inserción y cada eliminación tienen
que dejar en pie.

</Slide>

<Slide title="El inorden de un árbol de búsqueda">
**El inorden de un árbol binario de búsqueda entrega sus valores ordenados de
menor a mayor.**

La razón es la misma definición recursiva. El inorden de un nodo es el
inorden de su subárbol izquierdo, después el nodo, y después el inorden del
derecho. Por la invariante, todo lo de la izquierda es menor que el nodo y
todo lo de la derecha es mayor; si cada subárbol, más pequeño, ya sale
ordenado, el nodo queda exactamente entre los dos, y el total sale ordenado.

<SequenceStepper eda="binary-tree" operation="inorder" values={[4, 2, 6, 1, 3, 5, 7]} />

</Slide>

de búsqueda vacío: cada valor baja desde la raíz eligiendo un lado en cada
nodo, y se cuelga como hoja donde no hay más. Después escribamos el inorden
del árbol que queda.

</Slide>

<Slide title="Insertar a mano · la respuesta">
Cada valor quedó donde su búsqueda terminó, y el inorden, en la tira de

</Slide>

## Las operaciones del árbol de búsqueda

<Slide title="El árbol de búsqueda en Java">
La clase guarda la raíz y ofrece métodos públicos que delegan en uno
recursivo, que recibe la raíz de un subárbol. Para comparar dos valores se
usa `compareTo`, así que el tipo de los valores tiene que ser comparable.

```java
class BST<T extends Comparable<T>> {
    Node<T> root;

    boolean contains(T value) {
        return contains(root, value);
    }

    void insert(T value) {
        root = insert(root, value);
    }

    void remove(T value) {
        root = remove(root, value);
    }
}
```

`insert` y `remove` reasignan la raíz con lo que devuelve la versión
recursiva, porque insertar en un árbol vacío crea la raíz, y eliminarla
puede dejar otra en su lugar. Los nodos son los mismos `Node<T>` del árbol
binario.

</Slide>

<Slide title="Buscar en un árbol de búsqueda">
En cada nodo hay tres respuestas posibles: el valor es el buscado, es mayor,
o es menor. Si el buscado es menor, la invariante garantiza que no está a la
derecha, y se sigue solo por la izquierda; si es mayor, al revés. La
búsqueda baja por **un solo camino** y termina en el valor o en un subárbol
vacío.

<SequenceStepper eda="bst" operation="search" values={[4, 2, 7, 1, 3, 6, 9]} target={[6, 5]} />

Un camino desde la raíz tiene a lo más $$h + 1$$ nodos, así que buscar cuesta
$$O(h)$$, se encuentre o no el valor.

</Slide>

<Slide title="Insertar en un árbol de búsqueda">
Para insertar, se busca el valor. Si aparece, no se hace nada: el árbol no
guarda repetidos. Si la búsqueda termina en un subárbol vacío, ahí se cuelga
el valor nuevo como una **hoja**.

<SequenceStepper eda="bst" operation="insert" values={[8, 5, 1, 4, 6, 7, 20]} value={12} />

La invariante queda en pie: la hoja nueva quedó del lado que eligió cada
comparación, el lado correcto de cada nodo por encima de ella. Ningún otro
nodo se mueve, y el costo es el de la búsqueda: $$O(h)$$.

</Slide>

<Slide title="El mínimo y el máximo">
Por la invariante, todo lo menor que un nodo está a su izquierda. El
**mínimo** es entonces el nodo de más a la izquierda: se baja por la
izquierda hasta un nodo que no tenga hijo izquierdo. El **máximo** es el
simétrico: se baja por la derecha.

<SequenceStepper eda="bst" operation="minimum" values={[4, 2, 7, 1, 3, 6, 9]} />

Los dos bajan por un solo camino: $$O(h)$$. En un árbol vacío no hay mínimo,
y el método lanza una excepción.

</Slide>

<Slide title="El sucesor y el predecesor">
El **sucesor** de un valor es el menor valor del árbol que es mayor que él.
Se encuentra con una sola bajada desde la raíz: en cada nodo mayor que el
valor, ese nodo es un candidato y uno mejor solo puede estar a su izquierda;
en cada nodo menor o igual, el sucesor solo puede estar a su derecha. El
último candidato es el sucesor.

<SequenceStepper eda="bst" operation="successor" values={[4, 2, 7, 1, 3, 6, 9]} target={[3, 4]} />

Si el valor tiene subárbol derecho, el sucesor es el mínimo de ese subárbol;
si no, es el ancestro más cercano desde el que se bajó por la izquierda. El
**predecesor**, el mayor valor menor que él, es el simétrico. Los dos cuestan
$$O(h)$$.

</Slide>

<Slide title="Eliminar una hoja">
Para eliminar, primero se busca el valor. Cuando el nodo que lo tiene es una
**hoja**, basta con que su padre deje de apuntarle: la recursión devuelve el
subárbol vacío y el padre lo guarda en lugar del nodo.

<SequenceStepper eda="bst" operation="remove" values={[4, 2, 7, 1, 3, 6, 9]} target={1} />

</Slide>

<Slide title="Eliminar un nodo con un hijo">
Cuando el nodo tiene **un solo hijo**, su padre pasa a apuntar a ese hijo.

<SequenceStepper eda="bst" operation="remove" values={[8, 5, 1, 4, 6, 7, 20, 12]} target={1} />

</Slide>

La invariante se mantiene: todo el subárbol del hijo estaba del mismo lado
del padre que el nodo eliminado.

<Slide title="Eliminar un nodo con dos hijos">
Con **dos hijos**, el nodo toma el valor de su **sucesor** y se elimina el sucesor.

<SequenceStepper eda="bst" operation="remove" values={[8, 5, 1, 4, 6, 7, 20, 12]} target={8} />

</Slide>

Sacar el nodo dejaría dos subárboles colgando, y su padre solo tiene lugar
para uno; por eso se reemplaza su valor en lugar de quitarlo. El sucesor es el mínimo del subárbol derecho, y su valor es mayor que todo lo de la izquierda y menor que todo lo
demás de la derecha, así que la invariante queda en pie. Y el sucesor, por
ser el mínimo de un subárbol, no tiene hijo izquierdo: eliminarlo es uno de
los dos casos anteriores. Todo el trabajo es bajar dos veces por un camino:
$$O(h)$$.

<Slide title="Eliminar con dos hijos · la pregunta">
En este árbol, el 5 tiene dos hijos. Antes de pasar a la respuesta,
dibujemos cómo queda el árbol después de eliminarlo.

<BinaryPicture tree={{ v: 8, l: { v: 5, l: { v: 1, r: { v: 4 } }, r: { v: 6, r: { v: 7 } } }, r: { v: 20, l: { v: 12 } } }} tones={{ 5: 'focus' }} label="Un árbol de búsqueda: 8 en la raíz; a su izquierda el 5, con hijos 1 y 6; el 1 con un hijo derecho, 4; el 6 con un hijo derecho, 7; a la derecha del 8, el 20, con un hijo izquierdo, 12. El 5 está marcado." caption="eliminar 5, que tiene dos hijos" />

</Slide>

<Slide title="Eliminar con dos hijos · la respuesta">
El 5 toma el valor de su sucesor, 6, y el 6 de abajo deja su lugar al 7.

<SequenceStepper eda="bst" operation="remove" values={[8, 5, 1, 4, 6, 7, 20, 12]} target={5} />

</Slide>

<Slide title="Costos del árbol de búsqueda">
Cada operación baja por un camino desde la raíz, así que todas cuestan lo
mismo: la altura.

| Molde     | Operación                       | Peor caso  | Invariante que restaura                                    |
| --------- | ------------------------------- | ---------- | ---------------------------------------------------------- |
| Consultar | `contains`                      | $$O(h)$$   | Ninguna: no cambia el árbol                                |
| Consultar | mínimo, máximo                  | $$O(h)$$   | Ninguna: no cambia el árbol                                |
| Consultar | sucesor, predecesor             | $$O(h)$$   | Ninguna: no cambia el árbol                                |
| Modificar | `insert`                        | $$O(h)$$   | La de orden: la hoja nueva queda del lado que eligió cada comparación |
| Modificar | `remove`                        | $$O(h)$$   | La de orden: el sucesor cabe exactamente en el lugar que deja |

Y la altura de un árbol de $$n$$ nodos está entre $$\lfloor \log_2 n \rfloor$$
y $$n - 1$$.

</Slide>

<Slide title="Siete valores en un orden">
Los valores del 1 al 7, insertados en el orden 4, 2, 6, 1, 3, 5, 7: cada uno
cae en un nivel todavía incompleto, y el árbol queda lleno.

<SequenceStepper eda="bst" operation="insert" values={[]} value={[4, 2, 6, 1, 3, 5, 7]} />

</Slide>

<Slide title="Los mismos siete valores, ordenados">
Los mismos valores, insertados de menor a mayor: cada uno es mayor que todos
los anteriores, así que siempre baja por la derecha y se cuelga al final.

<SequenceStepper eda="bst" operation="insert" values={[]} value={[1, 2, 3, 4, 5, 6, 7]} />

El árbol quedó como una cadena de altura 6: buscar el 7 compara con todos
los nodos, igual que en una lista enlazada.

</Slide>

<Slide title="La altura que nadie controla">
El árbol de búsqueda responde cualquier pregunta sobre el orden —si un valor
está, el mínimo, el sucesor, insertar o eliminar— bajando por un solo camino:
cuesta la altura. Ninguna estructura de las que guardan elementos en fila
hace todo eso a la vez en tiempo logarítmico.

Pero la altura depende del **orden en que llegaron los valores**, y nada en el
árbol la controla. La invariante de orden dice dónde va cada valor y no dice
nada sobre la forma. Con los valores en el peor orden posible —ordenados—, el
árbol es una lista, y cada operación cuesta $$\Theta(n)$$.

Si las inserciones y las eliminaciones también cuidaran la **forma**,
reacomodando el árbol cada vez que se cargue hacia un lado, la altura quedaría
acotada por $$O(\log n)$$ y todas las operaciones con ella. El precio sería más
trabajo en cada inserción y en cada eliminación. Ese es el compromiso de un
**árbol de búsqueda balanceado**.

</Slide>

Un **ancestro común** de dos nodos tiene a los dos en su subárbol, y un nodo
cuenta en su propio subárbol. El **ancestro común más bajo** es el más
profundo de todos: en un árbol genealógico, el antepasado más cercano.

El programa recibe la raíz de un árbol binario de búsqueda y dos valores que
están en el árbol, y responde el valor de su ancestro común más bajo. Llamamos
$$h$$ a la altura del árbol.

```java
static int lowestCommonAncestor(Node<Integer> root,
        int a, int b)
```

| Valores  | Ancestro común más bajo |
| -------- | ----------------------- |
| 4 y 7    | 5                       |
| 6 y 7    | 6                       |
| 1 y 1    | 1                       |

<Slide title="La raíz como respuesta">
La raíz tiene a todos los nodos en su subárbol, así que siempre es un
ancestro común. Pero no siempre es el más bajo: para 4 y 7, el 8 es ancestro
de los dos, y el 5, más abajo, también lo es. Una respuesta correcta tiene
que saber **hasta dónde bajar** sin perder a ninguno de los dos.

</Slide>

<Slide title="Bajar mientras los dos estén del mismo lado">
Por la invariante del árbol de búsqueda, en cada nodo se sabe de qué lado
está cada valor. Si los dos son menores que el nodo, los dos están en su
subárbol izquierdo, y el ancestro común más bajo también: se baja a la
izquierda. Si los dos son mayores, a la derecha. En cuanto los dos **dejan de
estar del mismo lado** —uno a cada lado, o uno es el nodo mismo—, ningún nodo
de más abajo tiene a los dos: ese nodo es la respuesta.

Con 4 y 7: los dos son menores que 8, así que se baja al 5. El 4 es menor que
5 y el 7 es mayor: se separan, y el ancestro común más bajo es el 5.

</Slide>

<Slide title="Ancestro común · la solución">
Una sola bajada desde la raíz, sin recursión: en cada nodo, o los dos van
hacia el mismo lado, o se responde el nodo.

<StepShow
  language="java"
  code={`static int lowestCommonAncestor(Node<Integer> root,
        int a, int b) {
    Node<Integer> node = root;
    while (true) {
        if (a < node.value && b < node.value) {
            node = node.left;
        } else if (a > node.value && b > node.value) {
            node = node.right;
        } else {
            return node.value;
        }
    }
}`}

<Slide title="Un valor que es ancestro del otro">
Con 6 y 7, el ciclo baja del 8 al 5 y del 5 al 6. Ahí el 6 no es menor ni
mayor que el nodo: **es** el nodo, y la condición de bajar falla. El ciclo
responde 6, que es correcto, porque un nodo cuenta en su propio subárbol. Con
dos valores iguales, como 1 y 1, pasa lo mismo al llegar al 1.

Como los dos valores están en el árbol, la bajada siempre se detiene en un
nodo y nunca llega a un subárbol vacío.

**Tiempo.** Cada vuelta del ciclo baja un nivel, así que hay a lo más
$$h + 1$$ vueltas, y cada una hace a lo más cuatro comparaciones: $$O(h)$$.
**Memoria.** Una sola referencia, `node`, sin recursión: $$\Theta(1)$$.

</Slide>

```java starter
                node = node.left;
            } else if (a > node.value && b > node.value) {
```

<Slide title="Un arreglo de largo par y el arreglo vacío">
Con un largo par, como 1 2 3 4, no hay un único medio: la división entera
elige el de la izquierda, el 2. A su derecha quedan 3 y 4, que arman un
subárbol de altura 1, y el árbol queda de altura 2, que es
$$\lfloor \log_2 4 \rfloor$$: un lado queda un elemento más largo que el otro,
y eso nunca suma un nivel de más. Con el arreglo vacío, `lo` es 0 y `hi` es
$$-1$$: el primer caso devuelve `null` sin crear nada.

**Tiempo.** Cada llamada con un tramo no vacío crea exactamente un nodo, y
cada valor está en un solo tramo como medio: $$n$$ nodos, cada uno en tiempo
constante, $$\Theta(n)$$. **Memoria.** El árbol ocupa $$\Theta(n)$$; la
recursión, aparte, tiene la profundidad del árbol que arma,
$$\lfloor \log_2 n \rfloor + 1$$ llamadas: $$\Theta(\log n)$$.

</Slide>

<Slide title="Una calculadora con paréntesis">
Una calculadora recibe una expresión aritmética escrita con paréntesis y
tiene que devolver su valor. En la expresión, **cada operación va entre
paréntesis** con sus dos operandos, y cada operando es un número o, a su vez,
otra operación entre paréntesis.

<TokenRow tokens={CALC_CHARS} title="expresión" label="La expresión abre paréntesis, abre paréntesis, 3 más 4, cierra paréntesis, por 5, cierra paréntesis." />

El programa recibe la expresión como un `String` sin espacios, con números
enteros no negativos de uno o más dígitos y los operadores `+`, `-`, `*` y
`/` (la división es entera), y responde el valor como un `int`. La expresión
siempre está bien escrita. Llamamos $$N$$ a su largo.

```java
static int calculate(String expression)
```

| Expresión         | Valor |
| ----------------- | ----- |
| `((3+4)*5)`       | 35    |
| `(3+(4*5))`       | 23    |
| `((8-2)/(1+2))`   | 2     |
| `7`               | 7     |

</Slide>

<Slide title="Leer de izquierda a derecha">
`((3+4)*5)` y `(3+(4*5))` tienen los mismos números y los mismos operadores,
en el mismo orden. Si se operara de izquierda a derecha a medida que se lee,
las dos darían 35, y la segunda vale 23.

El orden en que se hacen las operaciones no está en el orden de los
caracteres: está en **qué paréntesis queda dentro de cuál**. Un programa que
lee la expresión una vez, de corrido, sin guardar ese anidamiento, no tiene
cómo saber qué operar primero.

</Slide>

<Slide title="Una expresión es un árbol">
Cada operación tiene un operador y dos operandos, y cada operando es un
número u otra operación. Es una jerarquía, y se dibuja como un **árbol
binario**: los operadores son nodos internos y los números son hojas. Es el
**árbol de expresión**.

<BinaryPicture tree={CALC_TREE} label="El árbol de la expresión abre paréntesis abre paréntesis 3 más 4 cierra por 5 cierra. La raíz es el por; su hijo izquierdo es el más, con hojas 3 y 4; su hijo derecho es la hoja 5." caption="((3+4)*5): el + queda debajo del *" />

Para armarlo se lee la expresión una vez: un paréntesis que abre anuncia un
operando izquierdo, un operador y un operando derecho, y cada operando se lee
con la misma regla. Lo que estaba dentro de otros paréntesis queda más abajo
en el árbol.

</Slide>

<Slide title="El postorden es la notación polaca inversa">
Para calcular un operador hace falta, antes, el valor de sus dos operandos:
el árbol se evalúa en **postorden**. Escrito, el postorden de un árbol de
expresión pone cada operador después de sus dos operandos, sin paréntesis: es
la **notación polaca inversa**.

<TokenRow tokens={['3', '4', '+', '5', '*']} title="postorden" label="El postorden del árbol: 3, 4, más, 5, por." />

Y la polaca inversa se evalúa con una **pila**, leyendo de izquierda a
derecha: cada número se apila, y cada operador saca los dos de arriba, los
opera y apila el resultado. Sobre `3 4 + 5 *`: se apilan 3 y 4; el `+` los
saca y apila 7; se apila 5; el `*` saca 7 y 5 y apila 35, que es el valor.

</Slide>

<Slide title="Calculadora · las tres piezas juntas">
`calculate` encadena las tres piezas: arma el árbol leyendo la expresión
desde la posición 0, lo recorre en postorden y evalúa la lista que queda.

```java
static int calculate(String expression) {
    pos = 0;
    Node<String> tree = parse(expression);
    List<String> tokens = new ArrayList<>();
    postorder(tree, tokens);
    return evaluate(tokens);
}
```

Sobre `((3+4)*5)`, el árbol tiene el `*` en la raíz, la lista es
`3 4 + 5 *` y la pila termina en 35.

<Slide title="Un número solo y una división por cero">
**Un número solo**, como `7`, no empieza con paréntesis: `parse` devuelve una
hoja, el postorden es la lista `7` y la pila apila el 7 y lo devuelve. No
hace falta ningún caso aparte.

**Una división por cero**, como `(4/(2-2))`, arma su árbol sin problema y
falla al evaluar: cuando el `/` saca 0 y 4 de la pila, `apply` divide por
cero y Java lanza `ArithmeticException`. El enunciado no dice qué hacer en
ese caso, así que la calculadora no lo oculta.

</Slide>

<Slide title="El costo de la calculadora">
**Tiempo.** `parse` avanza `pos` sobre cada carácter una sola vez, así que
arma el árbol en $$\Theta(N)$$. El árbol tiene a lo más $$N$$ nodos, porque
cada nodo sale de al menos un carácter; el postorden visita cada uno una vez
y la evaluación saca y apila cada valor de la lista una vez. Las tres piezas
cuestan $$\Theta(N)$$, y la calculadora también.

**Memoria.** El árbol, la lista y la pila guardan a lo más $$N$$ elementos
cada uno, y la recursión de `parse` y de `postorder` se apila tan hondo como
la altura del árbol, que no pasa de $$N$$. La memoria es $$\Theta(N)$$.

</Slide>

<Exercise title="Una calculadora con paréntesis">

```java starter
import java.util.ArrayList;
import java.util.List;
import java.util.Stack;

class Node<T> {
    T value;
    Node<T> left;
    Node<T> right;

    Node(T value) {
        this.value = value;
    }
}

class Solution {
    static int pos;

    static boolean isOperator(String t) {
        return t.equals("+") || t.equals("-")
            || t.equals("*") || t.equals("/");
    }

    static int apply(int a, int b, String t) {
        if (t.equals("+")) return a + b;
        if (t.equals("-")) return a - b;
        if (t.equals("*")) return a * b;
        return a / b;
    }

    static Node<String> parse(String s) {
        if (s.charAt(pos) != '(') {
            int start = pos;
            while (pos < s.length()
                    && Character.isDigit(s.charAt(pos))) {
                pos++;
            }
            return new Node<>(s.substring(start, pos));
        }
        pos++;
        Node<String> left = parse(s);
        String op = String.valueOf(s.charAt(pos));
        pos++;
        Node<String> right = parse(s);
        pos++;
        Node<String> node = new Node<>(op);
        node.left = left;
        node.right = right;
        return node;
    }

    static void postorder(Node<String> node,
            List<String> out) {
        if (node == null) return;
        postorder(node.left, out);
        postorder(node.right, out);
        out.add(node.value);
    }

    static int evaluate(List<String> tokens) {
        Stack<Integer> values = new Stack<>();
        for (String t : tokens) {
            if (isOperator(t)) {
                int b = values.pop();
                int a = values.pop();
                values.push(apply(a, b, t));
            } else {
                values.push(Integer.parseInt(t));
            }
        }
        return values.pop();
    }

    static int calculate(String expression) {
        pos = 0;
        Node<String> tree = parse(expression);
        List<String> tokens = new ArrayList<>();
        postorder(tree, tokens);
        return evaluate(tokens);
    }
}
```

```java test
check(Solution.calculate("((8-2)/(1+2))"), 2);
check(Solution.calculate("7"), 7);
check(Solution.calculate("(12*(10-7))"), 36);
```

```java solution
class Node<T> {
    T value;
    Node<T> left;
    Node<T> right;
    Node(T value) { this.value = value; }
}

class Solution {
    static boolean isBST(Node<Integer> root) {
        return isBST(root, Long.MIN_VALUE, Long.MAX_VALUE);
    }

    static boolean isBST(Node<Integer> node, long min, long max) {
        if (node == null) return true;
        if (node.value <= min || node.value >= max) return false;
        return isBST(node.left, min, node.value)
            && isBST(node.right, node.value, max);
    }
}
```
